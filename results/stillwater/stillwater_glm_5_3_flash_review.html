const isoDob = z
  .string()
  .refine((s) => /^\d{4}-\d{2}-\d{2}$/.test(s), "Use YYYY-MM-DD")
  .transform((s) => new Date(`${s}T00:00:00Z`));

function adultAge(dob: Date): number {
  const now = new Date();
  let age = now.getUTCFullYear() - dob.getUTCFullYear();
  const m = now.getUTCMonth() - dob.getUTCMonth();
  if (m < 0 || (m === 0 && now.getUTCDate() < dob.getUTCDate())) age--;
  return age;
}

// packages/db/src/crypto.ts
/**
 * App-level AES-256-GCM envelope encryption for health-adjacent payloads.
 *
 * Envelope format (all parts base64url):
 *   v1.<iv>.<ciphertext||authtag>
 *
 * The authtag is appended to the ciphertext blob rather than stored separately
 * so a single column carries the whole envelope.
 */
import { createCipheriv, createDecipheriv, randomBytes } from "node:crypto";
import "./env"; // validates ENCRYPTION_KEY presence

const VERSION = "v1";
const ALGO = "aes-256-gcm";
const IV_BYTES = 12;

let cachedKey: Buffer | null = null;

function key(): Buffer {
  if (!cachedKey) {
    const k = Buffer.from(process.env.ENCRYPTION_KEY!, "hex");
    if (k.length !== 32) {
      throw new Error("ENCRYPTION_KEY must decode to exactly 32 bytes.");
    }
    cachedKey = k;
  }
  return cachedKey;
}

export function encryptField(plaintext: string): string {
  const iv = randomBytes(IV_BYTES);
  const cipher = createCipheriv(ALGO, key(), iv);
  const ct = Buffer.concat([
    cipher.update(plaintext, "utf8"),
    cipher.final(),
    cipher.getAuthTag(),
  ]);
  return [
    VERSION,
    iv.toString("base64url"),
    ct.toString("base64url"),
  ].join(".");
}

export function decryptField(envelope: string): string {
  const [ver, ivPart, ctPart] = envelope.split(".");
  if (ver !== VERSION || !ivPart || !ctPart) {
    throw new Error("Malformed ciphertext envelope.");
  }
  const buf = Buffer.from(ctPart, "base64url");
  const tag = buf.subarray(buf.length - 16);
  const body = buf.subarray(0, buf.length - 16);
  const decipher = createDecipheriv(ALGO, key(), Buffer.from(ivPart, "base64url"));
  decipher.setAuthTag(tag);
  return Buffer.concat([decipher.update(body), decipher.final()]).toString("utf8");
}

export function looksLikeEnvelope(v: unknown): boolean {
  return typeof v === "string" && /^v1\.[A-Za-z0-9_-]+\.[A-Za-z0-9_-]+$/.test(v);
}

      this.prisma.client.memoryNote.findMany({
        where: { userId },
        orderBy: { createdAt: "desc" },
        take: MAX_NOTES,
      }),

        orderBy: { startedAt: "desc" },

      block += l + "\n";
      used += l.length;

/**
 * Deterministic crisis lexicon. Maintained by humans, reviewed quarterly.
 * Runs BEFORE and INDEPENDENT OF the model. Categories map to SafetyEvent.trigger.
 */
export type TriggerCategory =
  | "SUICIDAL_IDEATION"
  | "SELF_HARM"
  | "HARM_TO_OTHERS";

interface PatternDef {
  category: TriggerCategory;
  weight: number;
  re: RegExp;
}

const defs: PatternDef[] = [
  // ---- Suicidal ideation -------------------------------------------------
  { category: "SUICIDAL_IDEATION", weight: 5, re: /\bk(?:ill|m)s?\s+(?:my\s*)?(?:self)\b/i },
  { category: "SUICIDAL_IDEATION", weight: 5, re: /\bwanna\s+d[i1]\s*\b|\bdying\b.*\btired\b/i },
  { category: "SUICIDAL_IDEATION", weight: 5, re: /\bnot\s+worth\s+living\b/i },
  { category: "SUICIDAL_IDEATION", weight: 5, re: /\bbetter\s+off\s+(?:without\s+me|dead)\b/i },
  { category: "SUICIDAL_IDEATION", weight: 4, re: /\bkill myself\b/i },
  { category: "SUICIDAL_IDEATION", weight: 4, re: /\bsuicid(?:e|al)\b/i },
  { category: "SUICIDAL_IDEATION", weight: 4, re: /\bend\s+(?:it|my life|everything)\b/i },
  { category: "SUICIDAL_IDEATION", weight: 4, re: /\btake\s+my\s+(?:life|own\s+life)\b/i },
  { category: "SUICIDAL_IDEATION", weight: 3, re: /\bno reason to (?:go on|be alive|live)\b/i },
  { category: "SUICIDAL_IDEATION", weight: 3, re: /\bi (?:want|wish) (?:to die|i was dead)\b/i },

  // ---- Self-harm ----------------------------------------------------------
  { category: "SELF_HARM", weight: 4, re: /\bcut(?:t?ing)?\s+myself\b/i },
  { category: "SELF_HARM", weight: 4, re: /\bhurt(?:ing)?\s+myself\b/i },
  { category: "SELF_HARM", weight: 3, re: /\burn\b.*\bmy skin\b|\bpick(?:ing)?\s+at\s+my\s+skin\b/i },
  { category: "SELF_HARM", weight: 3, re: /\bstarving myself\b|\bmade myself sick\b/i },

  // ---- Harm to others -----------------------------------------------------
  { category: "HARM_TO_OTHERS", weight: 5, re: /\bhurt\s+(?:him|her|them|someone|somebody)\b/i },
  { category: "HARM_TO_OTHERS", weight: 5, re: /\bkill\s+(?:him|her|them|that guy|my boss)\b/i },
  { category: "HARM_TO_OTHERS", weight: 4, re: /\bfight(?:ing)?\s+(?:him|her|them)\s+(?:tonight|later|after)\b/i },
];

/** Normalise common evasions before scanning. */
export function normalise(raw: string): string {
  return raw
    .toLowerCase()
    .replace(/[0]/g, "o")
    .replace(/[$5]/g, "s")
    .replace(/[@]/g, "a")
    .replace(/\s+/g, " ")
    .trim();
}

export interface ScanHit {
  category: TriggerCategory;
  score: number;
}

/**
 * Score-based scan: strong phrases fire alone; weaker signals combine.
 * Threshold tuned high to favour recall on unmistakable intent while keeping
 * casual metaphor (“this meeting kills me”) below the line.
 */
const FIRE_THRESHOLD = 4;

export function scanTranscript(normalizedText: string): ScanHit | null {
  let best: ScanHit | null = null;
  let scores = new Map<TriggerCategory, number>();

  for (const d of defs) {
    if (d.re.test(normalizedText)) {
      const cur = (scores.get(d.category) ?? 0) + d.weight;
      scores.set(d.category, cur);
      if (!best || cur > best.score) best = { category: d.category, score: cur };
    }
  }
  if (best && best.score >= FIRE_THRESHOLD) return best;
  return null;
}

   * Evaluate a batch of freshly-finalised user turns. Purely deterministic —
   * identical input yields identical verdicts regardless of model behaviour.

    const hit = texts
      .map(normalise)
      .reduce<{ category: TriggerCategory; score: number } | null>((acc, t) => {

      data: {

@Injectable()
export class JwtAuthGuard implements CanActivate {
  constructor(private readonly jwt: JwtService) {}

  canActivate(ctx: ExecutionContext): boolean {
    const req = ctx.switchToHttp().getRequest();
    const header: string = req.headers.authorization ?? "";
    const [scheme, token] = header.split(" ");
    if (scheme !== "Bearer" || !token) throw new UnauthorizedException();

    try {
      const claims = this.jwt.verify(token) as RawJwtPayload;
      if ((claims as any).kind !== "access") throw new Error("wrong kind");
      req.user = { sub: claims.sub, email: claims.email };
      return true;
    } catch {
      throw new UnauthorizedException("Invalid or expired access token.");
    }
  }
}

    if (exists) throw new ConflictException("Email already registered.");

    const user = await this.prisma.client.user.create({
      data: {
        email: dto.email,
        passwordHash: await hash(dto.password),
        displayName: dto.displayName,
        dateOfBirth: dto.dateOfBirth,
        acceptedDisclosureAt: new Date(),
      },
    });
    return this.tokensFor(user);
  }

  async login(email: string, password: string) {
    const user = await this.prisma.client.user.findUnique({ where: { email } });
    if (!user || !(await verify(user.passwordHash, password))) {
      throw new UnauthorizedException("Wrong email or password.");
    }
    return this.tokensFor(user);
  }

  /** Rotating refresh: old token consumed atomically; reuse revokes the family. */
  async refresh(raw: string) {
    const rec = await this.prisma.client.refreshToken.findUnique({
      where: { tokenHash: sha256Hex(raw) },
      include: { user: true },
    });
    if (!rec) throw new UnauthorizedException();

    if (rec.revokedAt || rec.expiresAt < new Date()) {
      // Possible replay of a rotated token → burn everything for this user.
      await this.prisma.client.refreshToken.updateMany({
        where: { userId: rec.userId, revokedAt: null },
        data: { revokedAt: new Date() },
      });
      throw new UnauthorizedException("Please sign in again.");
    }

    await this.prisma.client.refreshToken.update({
      where: { id: rec.id },
      data: { revokedAt: new Date() },
    });
    return this.tokensFor(rec.user);
  }

  async logout(raw: string) {
    await this.prisma.client.refreshToken.updateMany({
      where: { tokenHash: sha256Hex(raw), revokedAt: null },
      data: { revokedAt: new Date() },
    });
  }

  private async tokensFor(user: { id: string; email: string }) {
    return {
      accessToken: this.accessToken(user),
      refreshToken: await this.issueRefresh(user.id),
      user: {
        id: user.id,
        email: user.email,
        displayName: user.displayName,
        preferredVoice: user.preferredVoice,
        focusWork: user.focusWork,
        acceptedDisclosureAt: user.acceptedDisclosureAt?.toISOString() ?? null,
      },
    };
  }

  assertAdult(dateOfBirth: Date) {
    const cutoff = new Date();
    cutoff.setUTCFullYear(cutoff.getUTCFullYear() - 18);
    if (dateOfBirth > cutoff) {
      throw new ForbiddenException(
        "Stillwater is for adults (18+). Please seek support suited to younger folks."
      );
    }
  }

  async findOrFail(id: string) {
    const u = await this.prisma.client.user.findUnique({ where: { id } });
    if (!u) throw new NotFoundException();
    return u;
  }
}

export interface Summarised {
  summary: string;
  themes: string[];
  mood: number;
}

const SUMMARISER_PROMPT = `
You summarise a finished voice-support conversation for the participant's own
future reference. Output STRICT JSON only:
{"summary": "<=70 words, warm third-person recap of what came up and any step forward",
 "themes": ["3 to 5 short noun phrases"],
 "mood": <-2..2 integer>}
Rules: no diagnoses, no advice, no sensitive identifiers beyond initials.
`;

export async function summarise(transcriptDigest: string): Promise<Summarised> {
  const ac = new AbortController();
  const kill = setTimeout(() => ac.abort(), 12_000); // keep T5 well under budget
  try {
    const resp = await client.chat.completions.create(
      {
        model: ENV.summaryModel,
        messages: [
          { role: "system", content: SUMMARISER_PROMPT },
          { role: "user", content: transcriptDigest },
        ],
        temperature: 0.3,
        response_format: { type: "json_object" },
      },
      { signal: ac.signal }
    );
    const raw = resp.choices[0]?.message?.content ?? "{}";
    const j = JSON.parse(raw) as Partial<Summarised>;
    return coerce(j);
  } finally {
    clearTimeout(kill);
  }
}

function clamp(n: number, lo: number, hi: number) {
  return Math.min(hi, Math.max(lo, n));
}

function coerce(j: Partial<Summarised>): Summarised {
  const themes = Array.isArray(j.themes)
    ? j.themes.filter((t) => typeof t === "string").slice(0, 5)
    : [];
  return {
    summary: typeof j.summary === "string" ? j.summary.slice(0, 600) : "",
    themes: themes.length >= 3 ? themes : [...themes, "reflection"].slice(0, 3),
    mood: clamp(Math.round(Number(j.mood ?? 0)), -2, 2),
  };
}

/** Fallback so ending a session never hangs on the provider. */
export function fallbackSummary(turnLines: string[]): Summarised {
  const joined = turnLines.join(" ").slice(0, 240);
  return {
    summary: joined ? `Talked about: ${joined}…` : "Short check-in.",
    themes: ["conversation"],
    mood: 0,
  };

  constructor(public status: number, public body: unknown) {
    super(`API ${status}`);
  }

let refreshingPromise: Promise<void> | null = null;

async function refreshOnce(): Promise<boolean> {
  const st = useAuth.getState();
  if (!st.refreshToken) return false;
  refreshingPromise ||= (async () => {
    try {
      const r = await fetch(`${BASE}/auth/refresh`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ refreshToken: st.refreshToken }),
      });
      if (!r.ok) throw new Error(String(r.status));
      const j = await r.json();
      st.setCredentials(j.accessToken, j.refreshToken);
    } finally {
      refreshingPromise = null;
    }
  })();
  return refreshingPromise.then(() => true).catch(() => false);
}

export async function apiFetch<T>(
  path: string,
  init: RequestInit & { auth?: boolean } = {}
): Promise<T> {
  const { auth = true, ...rest } = init;
  const token = useAuth.getState().accessToken;

  const doCall = () =>
    fetch(`${BASE}${path}`, {
      ...rest,
      headers: {
        ...(rest.body ? { "Content-Type": "application/json" } : {}),
        ...(auth && token ? { Authorization: `Bearer ${token}` } : {}),
        ...rest.headers,
      },
    });

  let res = await doCall();
  if (res.status === 401 && auth && (await refreshOnce())) {
    res = await doCall();
  }
  const body = res.status === 204 ? null : await res.json().catch(() => null);
  if (!res.ok) throw new ApiError(res.status, body);
  return body as T;
}

    setCaptions((prev) => [...prev.slice(-40), { id: ++captionSeq, role, text }]);
  }, []);

  /* ---------------- outbound turn batching + safety ---------------------- */
  const flushBuffer = useRef<NodeJS.Timeout>();
  const queueRef = useRef<{ role: "user" | "assistant"; text: string; startedAtMs: number }[]>([]);

  const enqueueTurn = useCallback((role: "user" | "assistant", text: string) => {
    queueRef.current.push({ role, text, startedAtMs: Date.now() });
    digestRef.current.push(`${role === "user" ? "Them" : "Companion"}: ${text}`);
    clearTimeout(flushBuffer.current);
    flushBuffer.current = setTimeout(async () => {
      const batch = queueRef.current.splice(0);
      if (!batch.length) return;
      try {
        const res = await apiFetch<TurnAnalysisResult>(`/sessions/${sessionId}/turns`, {
          method: "POST",
          body: JSON.stringify({ turns: batch }),
        });
        if (res.flagged) {
          setAnalysis(res);
          applyServerInstructionPatch(res.instructionPatch);
        }
      } catch {
        /* offline tolerance: turns retried opportunistically on next flush */
        queueRef.current.unshift(...batch);
      }
    }, 350);
  }, [sessionId]);

  const sendDC = useCallback((obj: unknown) => {
    dcRef.current?.send(JSON.stringify(obj));
  }, []);

  const applyServerInstructionPatch = useCallback((patch?: string) => {
    if (!patch) return;
    sendDC({ type: "session.update", session: { instructions: patch, temperature: 0.4 } });
  }, [sendDC]);

  /* ---------------- signalling handshake --------------------------------- */
  const negotiate = useCallback(async (cred: EphemeralCredential) => {
    const pc = new RTCPeerConnection({ iceServers: cred.iceServers });
    pcRef.current = pc;

    pc.addEventListener("connectionstatechange", () => {
      if (pc.connectionState === "connected") setPhase("live");
      if (pc.connectionState === "failed") failOver("network lost");
    });

    const stream = await mediaDevices.getUserMedia({
      audio: { echoCancellation: true, noiseSuppression: true, autoGainControl: true },
      video: false,
    });
    trackRef.current = stream.getTracks()[0];
    stream.getAudioTracks().forEach((t) => pc.addTrack(t, stream));

    const dc = pc.createDataChannel("oai-events");
    dcRef.current = dc;
    dc.onopen = () => {
      sendDC({
        type: "session.update",
        session: {
          tools: [], // none — pure conversation
          tool_choice: "auto",
        },
      });
    };
    dc.onmessage = (ev) => handleServerEvent(ev.data);

    const offer = await pc.createOffer({ offerToReceiveAudio: true });
    await pc.setLocalDescription(offer);

    // Await trickle-complete: wait for gathering to settle (small delay suffices
    // because STUN-less host candidates dominate on LAN/Wi-Fi setups).
    await waitForIceGathering(pc, 1500);

    const resp = await fetch(`${REALTIME_URL}?model=${encodeURIComponent(cred.model)}`, {
      method: "POST",
      headers: {
        Authorization: `Bearer ${cred.clientSecret}`,
        "Content-Type": "application/sdp",
      },
      body: pc.localDescription!.sdp,
    });
    if (!resp.ok) throw new Error(`SDP exchange failed (${resp.status})`);

    const answerSdp = await resp.text();
    await pc.setRemoteDescription({ type: "answer", sdp: answerSdp } as RTCSessionDescription);

    pc.addEventListener("track", (evt: any) => {
      const inbound = evt.streams[0] ?? evt.track;
      void inbound; // playback occurs implicitly once remote description lands
    });
  }, [pushCaption, sendDC]);

  const waitForIceGathering = (pc: InstanceType<typeof RTCPeerConnection>, maxMs: number) =>
    new Promise<void>((resolve) => {
      if (pc.iceGatheringState === "complete") return resolve();
      const t = setTimeout(resolve, maxMs);
      pc.addEventListener("icegatheringstatechange", () => {
        if (pc.iceGatheringState === "complete") { clearTimeout(t); resolve(); }
      });
    });

  /* ---------------- server-event routing --------------------------------- */
  const handleServerEvent = useCallback((raw: string) => {
    let ev: any;
    try { ev = JSON.parse(raw); } catch { return; }

    switch (ev.type) {
      case "input_audio_buffer.speech_started":
        setOrbMode("listening"); // user grabbed the floor
        break;
      case "response.audio_transcript.delta":
        assistantPartial.current += ev.delta ?? "";
        break;
      case "response.audio_transcript.done": {
        const txt = (ev.transcript ?? assistantPartial.current).trim();
        assistantPartial.current = "";
        if (txt) { pushCaption("assistant", txt); enqueueTurn("assistant", txt); }
        setOrbMode(muted ? "muted" : "idle");
        break;
      }
      case "conversation.item.input_audio_transcription.completed":
        pendingUserBuf.current = (ev.transcript ?? "").trim();
        break;
      case "response.created":
        setOrbMode("speaking");
        break;
      case "response.done":
        if (pendingUserBuf.current) {
          pushCaption("user", pendingUserBuf.current);
          enqueueTurn("user", pendingUserBuf.current);
          pendingUserBuf.current = "";
        }
        break;
      case "error":
        if (ev.error?.code === "invalid_api_key" || ev.error?.code === "expired_token") {
          rotateCredentialSilently();
        }
        break;
    }
  }, [enqueueTurn, muted, pushCaption]);

  /* ---------------- credential lifecycle (T8) ---------------------------- */
  const rotateCredentialSilently = useCallback(async () => {
    const prev = credRef.current;
    try {
      const fresh = await apiFetch<EphemeralCredential>("/realtime/token", {
        method: "POST",
        body: JSON.stringify({ voice: useAuth.getState().preferredVoice }),
      });
      credRef.current = fresh;
      scheduleRenewal(fresh);
      // Seamless swap: renegotiate on the SAME PeerConnection where possible.
      await renegotiateWithFreshSecret(fresh);
    } catch (e) {
      if (!endedByUs.current) failOver("credential renewal failed");
    }
    void prev;
  }, []);

  const renegotiateWithFreshSecret = async (fresh: EphemeralCredential) => {
    const pc = pcRef.current!;
    const offer = await pc.createOffer({ offerToReceiveAudio: true });
    await pc.setLocalDescription(offer);
    await waitForIceGathering(pc, 1200);
    const resp = await fetch(`${REALTIME_URL}?model=${encodeURIComponent(fresh.model)}`, {
      method: "POST",
      headers: { Authorization: `Bearer ${fresh.clientSecret}`, "Content-Type": "application/sdp" },
      body: pc.localDescription!.sdp,
    });
    if (!resp.ok) throw new Error("renegotiation rejected");
    await pc.setRemoteDescription({ type: "answer", sdp: await resp.text() } as RTCSessionDescription);
  };

  const scheduleRenewal = (cred: EphemeralCredential) => {
    clearTimeout(renewTimer.current);
    const lead = 30_000; // renew half a minute before expiry
    const msLeft = Math.max(5_000, new Date(cred.expiresAtIso).getTime() - Date.now() - lead);
    renewTimer.current = setTimeout(rotateCredentialSilently, msLeft);
  };

  /* ---------------- lifecycle controls ----------------------------------- */
  const start = useCallback(async () => {
    try {
      setPhase("connecting");
      setOrbMode("connecting");
      const cred = await apiFetch<EphemeralCredential>("/realtime/token", {
        method: "POST",
        body: JSON.stringify({ voice: useAuth.getState().preferredVoice }),
      });
      credRef.current = cred;
      scheduleRenewal(cred);
      await negotiate(cred);
    } catch (e: any) {
      failOver(e?.message ?? "could not start");
    }
  }, [negotiate]);

  const failOver = useCallback((reason: string) => {
    if (endedByUs.current) return;
    setPhase("failed");
    setOrbMode("muted");
    teardownPeerOnly();
    onComplete({ ok: false, interrupted: true });
  }, [onComplete]);

  const toggleMute = useCallback(() => {
    const next = !muted;
    setMuted(next);
    trackRef.current?.setEnabled(!next);
    setOrbMode(next ? "muted" : phase === "live" ? "idle" : orbMode);
  }, [muted, phase, orbMode]);

  const teardownPeerOnly = useCallback(() => {
    clearTimeout(renewTimer.current);
    clearInterval(flushBuffer.current);
    try { dcRef.current?.close(); } catch {}
    try { pcRef.current?.close(); } catch {}
    trackRef.current?.stop();
    dcRef.current = null; pcRef.current = null; trackRef.current = null;
  }, []);

  const endGracefully = useCallback(async (markInterrupted: boolean) => {
    endedByUs.current = true;
    teardownPeerOnly();
    setPhase("disconnected");
    try {
      await apiFetch(`/sessions/${sessionId}`, {
        method: "PATCH",
        body: JSON.stringify({ status: markInterrupted ? "INTERRUPTED" : "COMPLETED" }),
      });
      await apiFetch(`/sessions/${sessionId}/finish`, {
        method: "POST",
        body: JSON.stringify({ digest: digestRef.current.slice(-60) }),
      });
      onComplete({ ok: true, interrupted: markInterrupted });
    } catch {
      onComplete({ ok: false, interrupted: true });
    }
  }, [sessionId, onComplete, teardownPeerOnly]);

  useEffect(() => () => { endedByUs.current = true; teardownPeerOnly(); }, []);

  return {
    phase, orbMode, captions, muted, analysis,
    start, endGracefully, toggleMute, dismissAnalysis: () => setAnalysis(null),
  };
}

        if (/permission|NotAllowed/i.test(e?.message ?? "")) setMicDenied(true);
        else setNetLost(true);
      }
    })();
  }, []);

  /* Backgrounding: mute + freeze visuals; prolonged absence ends politely. */
  useEffect(() => {
    const sub = AppState.addEventListener("change", (s) => {
      if (s === "background" && !sess.muted) sess.toggleMute();
    });
    return () => sub.remove();
  }, [sess.muted]);

  /* Hardware back shouldn't yank the rug out mid-conversation. */
  useEffect(() => {
    const bh = BackHandler.addEventListener("hardwareBackPress", () => true);
    return () => bh.remove();
  }, []);

  const orbMode: OrbMode =
    micDenied || netLostBanner ? "muted" : sess.orbMode;

  return (
    <SafeAreaView className="flex-1 bg-ink-950">
      {/* Header strip */}
      <View className="flex-row items-center justify-between px-6 pt-2">
        <Text className="text-xs uppercase tracking-widest text-slate-500">
          {sess.phase === "live" ? "Listening" : sess.phase === "connecting" ? "Connecting…" : sess.phase}
        </Text>
        <Pressable onPress={sess.toggleMute} className="rounded-full border border-white/10 px-4 py-1.5">
          <Text className="text-xs text-slate-300">{sess.muted ? "🔇 Unmute" : "🎙 Mute"}</Text>
        </Pressable>
      </View>

      {(micDenied || netLostBanner) && (
        <View className="mx-4 mt-3 rounded-xl bg-amber-900/60 p-3">
          <Text className="text-xs text-amber-100">
            {micDenied
              ? "We need microphone access. Enable it in iOS Settings → Stillwater."
              : "Can't reach the session. Check your connection — your progress so far is saved."}
          </Text>
          {micDenied && (
            <Pressable onPress={() => Linking.openSettings()}>
              <Text className="mt-1 text-xs font-semibold text-amber-300 underline">Open settings</Text>
            </Pressable>
          )}
        </View>
      )}

      {/* The orb */}
      <View className="flex-none items-center justify-center grow">
        <Orb mode={orbMode} />
      </View>

      {/* Crisis overlay takes absolute priority visually */}
      {sess.analysis?.flagged && (
        <CrisisCard
          analysis={sess.analysis}
          onAcknowledged={async () => {
            try {
              await apiFetch("/safety/acknowledge-card", {
                method: "POST",
                body: JSON.stringify({ eventId: sess.analysis!.eventId }),
              });
            } finally { sess.dismissAnalysis(); }
          }}
        />
      )}

      {/* Captions */}
      <View className="h-56 pb-2">
        <CaptionsFeed lines={sess.captions} />
      </View>

      {/* Sole exit affordance */}
      <View className="px-6 pb-6">
        <Pressable
          onPress={() => sess.endGracefully(netLostBanner)}
          className="items-center rounded-2xl border border-white/15 py-4 active:border-white/40"
        >
          <Text className="font-medium text-slate-200">■  End session</Text>
        </Pressable>
      </View>
    </SafeAreaView>
  );
}

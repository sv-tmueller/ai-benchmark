```js
// src/core/Renderer.js
import * as THREE from 'three';

export default class Renderer {
  constructor({ canvas }) {
    this.renderer = new THREE.WebGLRenderer({ canvas, antialias: true, alpha: false });
    this.renderer.setPixelRatio(window.devicePixelRatio);
    this.renderer.setSize(window.innerWidth, window.innerHeight);
    this.renderer.shadowMap.enabled = true;
    this.renderer.shadowMap.type = THREE.PCFSoftShadowMap;

    window.addEventListener('resize', () => this.onResize());
  }

  onResize() {
    const { innerWidth, innerHeight } = window;
    this.renderer.setSize(innerWidth, innerHeight);
    this.camera.aspect = innerWidth / innerHeight;
    this.camera.updateProjectionMatrix();
  }

  setCamera(camera) {
    this.camera = camera;
  }

  render(scene) {
    this.renderer.render(scene, this.camera);
  }
}
```

### 3.3 InputManager.js
```js
// src/core/InputManager.js
export default class InputManager {
  constructor() {
    this.keys = {};
    this.gamepad = null;
    this.deadzone = 0.15;

    window.addEventListener('keydown', (e) => (this.keys[e.code] = true));
    window.addEventListener('keyup', (e) => (this.keys[e.code] = false));
    window.addEventListener('gamepadconnected', (e) => (this.gamepad = e.gamepad));
    window.addEventListener('gamepaddisconnected', () => (this.gamepad = null));
  }

  // Returns a normalized vector {x, y} where x = steering, y = throttle/brake
  getAxis() {
    // Keyboard fallback
    const steer = (this.keys['ArrowLeft'] || this.keys['KeyA'] ? -1 : 0) +
                  (this.keys['ArrowRight'] || this.keys['KeyD'] ? 1 : 0);
    const accel = (this.keys['ArrowUp'] || this.keys['KeyW'] ? 1 : 0) +
                  (this.keys['ArrowDown'] || this.keys['KeyS'] ? -1 : 0);

    // Gamepad override
    if (this.gamepad) {
      const gp = navigator.getGamepads()[this.gamepad.index];
      if (gp) {
        const leftX = Math.abs(gp.axes[0]) > this.deadzone ? gp.axes[0] : 0;
        const leftY = Math.abs(gp.axes[1]) > this.deadzone ? -gp.axes[1] : 0; // invert Y
        return { steer: leftX, accel: leftY };
      }
    }

    return { steer, accel };
  }

  // Button queries (e.g., for items, drift)
  isPressed(code) {
    return this.keys[code] || false;
  }
}
```

```js
export default class Time {
  constructor() {
    this.last = performance.now();
    this.delta = 0;
    this.elapsed = 0;
  }

  tick() {
    const now = performance.now();
    this.delta = (now - this.last) / 1000; // seconds
    this.last = now;
    this.elapsed += this.delta;
  }
}
```

### 3.5 Physics.js (Ammo.js wrapper)
```js
// src/core/Physics.js
import * as THREE from 'three';
import Ammo from 'ammo.js';

export default class Physics {
  constructor() {
    this.collisionConfiguration = new Ammo.btDefaultCollisionConfiguration();
    this.dispatcher = new Ammo.btCollisionDispatcher(this.collisionConfiguration);
    this.broadphase = new Ammo.btDbvtBroadphase();
    this.solver = new Ammo.btSequentialImpulseConstraintSolver();
    this.world = new Ammo.btDiscreteDynamicsWorld(
      this.dispatcher,
      this.broadphase,
      this.solver,
      this.collisionConfiguration
    );
    this.world.setGravity(new Ammo.btVector3(0, -9.81, 0));
    this.rigidBodies = [];
    this.tmpTransform = new Ammo.btTransform();
  }

  // Helper: convert THREE.Vector3 → Ammo.btVector3 (reused)
  static threeToAmmo(v) {
    return new Ammo.btVector3(v.x, v.y, v.z);
  }

  // Add a rigid body (mesh + shape)
  addRigidBody(mesh, shape, mass = 0, friction = 0.8, restitution = 0.1) {
    const transform = new Ammo.btTransform();
    transform.setIdentity();
    const pos = mesh.position;
    transform.setOrigin(Physics.threeToAmmo(pos));
    const quat = mesh.quaternion;
    transform.setRotation(new Ammo.btQuaternion(quat.x, quat.y, quat.z, quat.w));

    const motionState = new Ammo.btDefaultMotionState(transform);
    const localInertia = new Ammo.btVector3(0, 0, 0);
    if (mass > 0) shape.calculateLocalInertia(mass, localInertia);

    const rbInfo = new Ammo.btRigidBodyConstructionInfo(mass, motionState, shape, localInertia);
    const body = new Ammo.btRigidBody(rbInfo);
    body.setFriction(friction);
    body.setRestitution(restitution);
    this.world.addRigidBody(body);

    mesh.userData.physicsBody = body;
    if (mass > 0) this.rigidBodies.push(mesh);
    return body;
  }

  // Step simulation and sync Three.js meshes
  step(delta) {
    this.world.stepSimulation(delta, 10);
    for (const mesh of this.rigidBodies) {
      const body = mesh.userData.physicsBody;
      const ms = body.getMotionState();
      if (ms) {
        ms.getWorldTransform(this.tmpTransform);
        const p = this.tmpTransform.getOrigin();
        const q = this.tmpTransform.getRotation();
        mesh.position.set(p.x(), p.y(), p.z());
        mesh.quaternion.set(q.x(), q.y(), q.z(), q.w());
      }
    }
  }
}
```

> **Why Ammo.js?**  
> It’s a WebAssembly‑compiled port of Bullet Physics, offering **real‑time vehicle dynamics** (ray‑cast wheels) and **high‑performance collision** while staying fully compatible with Three.js.

### 4.1 Kart.js – Player & AI Base

```js
// src/game/Kart.js
import * as THREE from 'three';
import { RaycastVehicle } from 'three/examples/jsm/objects/Vehicle.js';
import Physics from '../core/Physics.js';
import InputManager from '../core/InputManager.js';
import { CONFIG } from '../config.js';

export default class Kart {
  constructor({ scene, physics, input, modelUrl, position = new THREE.Vector3() }) {
    this.scene = scene;
    this.physics = physics;
    this.input = input;
    this.modelUrl = modelUrl;
    this.position = position.clone();

    this.maxSpeed = CONFIG.KART_MAX_SPEED; // m/s
    this.acceleration = CONFIG.KART_ACCEL;
    this.brakeForce = CONFIG.KART_BRAKE;
    this.steerAngle = CONFIG.KART_STEER_ANGLE; // radians
    this.driftFactor = CONFIG.KART_DRIFT_FACTOR;
    this.boostMultiplier = 1;
    this.isBoosting = false;
    this.boostTimer = 0;

    this.loadModel();
  }

  async loadModel() {
    const loader = new THREE.GLTFLoader();
    const gltf = await loader.loadAsync(this.modelUrl);
    this.mesh = gltf.scene;
    this.mesh.traverse((c) => {
      if (c.isMesh) c.castShadow = true;
    });
    this.mesh.position.copy(this.position);
    this.scene.add(this.mesh);

    // ---- Physics ----
    // Simple chassis shape (box) – replace with convex hull for higher fidelity
    const chassisShape = new Ammo.btBoxShape(new Ammo.btVector3(0.8, 0.5, 1.2));
    const mass = CONFIG.KART_MASS;
    this.chassisBody = this.physics.addRigidBody(this.mesh, chassisShape, mass);
    this.chassisBody.setActivationState(4); // DISABLE_DEACTIVATION

    // ---- Vehicle (ray‑cast wheels) ----
    const tuning = new Ammo.btVehicleTuning();
    const rayCaster = new Ammo.btDefaultVehicleRaycaster(this.physics.world);
    this.vehicle = new Ammo.btRaycastVehicle(tuning, this.chassisBody, rayCaster);
    this.vehicle.setCoordinateSystem(0, 1, 2); // right, up, forward
    this.physics.world.addAction(this.vehicle);

    // wheel parameters (front left, front right, rear left, rear right)
    const wheelInfo = [
      { isFront: true,  pos: new THREE.Vector3(-0.8, -0.4, 1.2) },
      { isFront: true,  pos: new THREE.Vector3( 0.8, -0.4, 1.2) },
      { isFront: false, pos: new THREE.Vector3(-0.8, -0.4, -1.2) },
      { isFront: false, pos: new THREE.Vector3( 0.8, -0.4, -1.2) },
    ];

    const wheelRadius = 0.35;
    const wheelWidth = 0.2;
    const suspensionRestLength = 0.6;
    const suspensionStiffness = 20;
    const suspensionDamping = 2.3;
    const suspensionCompression = 4.4;
    const frictionSlip = 1000;
    const rollInfluence = 0.01;

    wheelInfo.forEach((info) => {
      const wheel = this.vehicle.addWheel(
        new Ammo.btVector3(info.pos.x, info.pos.y, info.pos.z),
        new Ammo.btVector3(0, -1, 0), // wheel direction
        new Ammo.btVector3(-1, 0, 0), // wheel axle
        suspensionRestLength,
        wheelRadius,
        tuning,
        info.isFront
      );
      wheel.setSuspensionStiffness(suspensionStiffness);
      wheel.setWheelsDampingRelaxation(suspensionDamping);
      wheel.setWheelsDampingCompression(suspensionCompression);
      wheel.setFrictionSlip(frictionSlip);
      wheel.setRollInfluence(rollInfluence);
    });

    // Visual wheel meshes (optional but nice)
    this.wheelMeshes = [];
    const wheelGeom = new THREE.CylinderGeometry(wheelRadius, wheelRadius, wheelWidth, 24);
    const wheelMat = new THREE.MeshStandardMaterial({ color: 0x222222 });
    for (let i = 0; i < 4; i++) {
      const wm = new THREE.Mesh(wheelGeom, wheelMat);
      wm.rotation.z = Math.PI / 2;
      this.scene.add(wm);
      this.wheelMeshes.push(wm);
    }
  }

  // -----------------------------------------------------------------
  //  Update – called each frame from GameManager
  // -----------------------------------------------------------------
  update(dt) {
    if (!this.vehicle) return; // model still loading

    // ---- Input handling ----
    const { steer, accel } = this.input.getAxis();

    // Steering
    const steering = THREE.MathUtils.clamp(steer, -1, 1) * this.steerAngle;
    this.vehicle.setSteeringValue(steering, 0); // front left
    this.vehicle.setSteeringValue(steering, 1); // front right

    // Acceleration / braking
    const engineForce = accel > 0
      ? accel * this.acceleration * this.boostMultiplier
      : accel < 0
        ? accel * this.brakeForce
        : 0;
    this.vehicle.applyEngineForce(engineForce, 2); // rear left
    this.vehicle.applyEngineForce(engineForce, 3); // rear right

    // ---- Drift (simple lateral slip reduction) ----
    if (Math.abs(steer) > 0.7 && Math.abs(accel) > 0.2) {
      // apply a temporary boost when drifting
      this.startBoost(CONFIG.DRIFT_BOOST_TIME, CONFIG.DRIFT_BOOST_MULT);
    }

    // ---- Boost timer ----
    if (this.isBoosting) {
      this.boostTimer -= dt;
      if (this.boostTimer <= 0) {
        this.isBoosting = false;
        this.boostMultiplier = 1;
      }
    }

    // ---- Sync visual wheels ----
    for (let i = 0; i < this.vehicle.getNumWheels(); i++) {
      this.vehicle.updateWheelTransform(i, true);
      const tm = this.vehicle.getWheelTransformWS(i);
      const p = tm.getOrigin();
      const q = tm.getRotation();
      const wheelMesh = this.wheelMeshes[i];
      wheelMesh.position.set(p.x(), p.y(), p.z());
      wheelMesh.quaternion.set(q.x(), q.y(), q.z(), q.w());
    }

    // ---- Update HUD (speed) ----
    const velocity = this.chassisBody.getLinearVelocity();
    const speed = Math.sqrt(
      velocity.x() * velocity.x() +
      velocity.y() * velocity.y() +
      velocity.z() * velocity.z()
    );
    this.speedKmh = speed * 3.6; // m/s → km/h
  }

  // -----------------------------------------------------------------
  //  Boost handling (item, drift, etc.)
  // -----------------------------------------------------------------
  startBoost(durationSec, multiplier) {
    this.isBoosting = true;
    this.boostTimer = durationSec;
    this.boostMultiplier = multiplier;
  }

  // -----------------------------------------------------------------
  //  Simple collision callback for items / checkpoints
  // -----------------------------------------------------------------
  onCollision(other) {
    // placeholder – you can expand to handle item boxes, walls, etc.
    if (other.userData.type === 'ItemBox') {
      // give player a random item
      this.scene.dispatchEvent({ type: 'itemPickup', kart: this });
    }
  }
}
```

### 4.2 AIKart.js – Simple Path‑Following AI

```js
// src/game/AIKart.js
import * as THREE from 'three';
import Kart from './Kart.js';
import { CONFIG } from '../config.js';

export default class AIKart extends Kart {
  constructor(opts) {
    super(opts);
    this.path = opts.path; // THREE.Curve (the same spline used for the track)
    this.pathProgress = 0; // 0‑1 along the spline
    this.speedTarget = CONFIG.AI_BASE_SPEED;
  }

  update(dt) {
    if (!this.vehicle) return;

    // Move along the spline
    this.pathProgress = (this.pathProgress + this.speedTarget * dt / this.path.getLength()) % 1;
    const targetPos = this.path.getPointAt(this.pathProgress);
    const targetTangent = this.path.getTangentAt(this.pathProgress);

    // Compute steering angle needed to face the tangent
    const forward = new THREE.Vector3(0, 0, 1).applyQuaternion(this.mesh.quaternion);
    const desired = targetTangent.clone().setY(0).normalize();
    const current = forward.clone().setY(0).normalize();
    const angleDiff = Math.atan2(
      desired.x * current.z - desired.z * current.x,
      desired.x * current.x + desired.z * current.z
    );
    const steer = THREE.MathUtils.clamp(angleDiff / this.steerAngle, -1, 1);

    // Simple throttle logic – accelerate if we’re behind target speed
    const velocity = this.chassisBody.getLinearVelocity();
    const speed = Math.sqrt(velocity.x() ** 2 + velocity.z() ** 2);
    const accel = speed < this.speedTarget ? 1 : 0;

    // Apply to base class (reuse its physics handling)
    this.vehicle.setSteeringValue(steer * this.steerAngle, 0);
    this.vehicle.setSteeringValue(steer * this.steerAngle, 1);
    this.vehicle.applyEngineForce(accel * this.acceleration, 2);
    this.vehicle.applyEngineForce(accel * this.acceleration, 3);

    // Sync visual wheels (same as parent)
    super.update(dt);
  }
}
```

### 4.3 Track.js – Spline‑Based Track Generation

```js
// src/game/Track.js
import * as THREE from 'three';
import { GLTFLoader } from 'three/examples/jsm/loaders/GLTFLoader.js';
import Physics from '../core/Physics.js';
import { CONFIG } from '../config.js';

export default class Track {
  constructor({ scene, physics, splinePoints }) {
    this.scene = scene;
    this.physics = physics;
    this.spline = new THREE.CatmullRomCurve3(splinePoints, true, 'catmullrom', 0.5);
    this.trackMesh = null;
    this.checkpoints = []; // array of {position, radius}
    this.itemBoxes = [];

    this.buildMesh();
    this.buildCollision();
    this.placeCheckpoints();
    this.placeItemBoxes();
  }

  // -----------------------------------------------------------------
  //  Build the visual track (extruded shape)
  // -----------------------------------------------------------------
  buildMesh() {
    const shape = new THREE.Shape();
    shape.moveTo(-CONFIG.TRACK_WIDTH / 2, 0);
    shape.lineTo(CONFIG.TRACK_WIDTH / 2, 0);
    shape.lineTo(CONFIG.TRACK_WIDTH / 2, CONFIG.TRACK_LENGTH);
    shape.lineTo(-CONFIG.TRACK_WIDTH / 2, CONFIG.TRACK_LENGTH);
    shape.closePath();

    const extrudeSettings = {
      steps: 200,
      bevelEnabled: false,
      extrudePath: this.spline,
    };
    const geometry = new THREE.ExtrudeGeometry(shape, extrudeSettings);
    geometry.computeVertexNormals();

    const material = new THREE.MeshStandardMaterial({
      color: 0x555555,
      metalness: 0.2,
      roughness: 0.8,
    });
    this.trackMesh = new THREE.Mesh(geometry, material);
    this.trackMesh.receiveShadow = true;
    this.scene.add(this.trackMesh);
  }

  // -----------------------------------------------------------------
  //  Collision shape (static compound) – cheap but works
  // -----------------------------------------------------------------
  buildCollision() {
    // Approximate track with a series of thin boxes along the spline
    const segmentCount = 100;
    const segmentLength = this.spline.getLength() / segmentCount;
    const halfWidth = CONFIG.TRACK_WIDTH / 2;

    const boxShape = new Ammo.btBoxShape(new Ammo.btVector3(halfWidth, 0.5, segmentLength / 2));
    for (let i = 0; i < segmentCount; i++) {
      const t0 = i / segmentCount;
      const t1 = (i + 1) / segmentCount;
      const p0 = this.spline.getPointAt(t0);
      const p1 = this.spline.getPointAt(t1);
      const mid = new THREE.Vector3().addVectors(p0, p1).multiplyScalar(0.5);
      const dir = new THREE.Vector3().subVectors(p1, p0).normalize();
      const quat = new THREE.Quaternion().setFromUnitVectors(
        new THREE.Vector3(0, 0, 1), // default forward
        dir
      );

      const transform = new Ammo.btTransform();
      transform.setIdentity();
      transform.setOrigin(new Ammo.btVector3(mid.x, mid.y, mid.z));
      transform.setRotation(new Ammo.btQuaternion(quat.x, quat.y, quat.z, quat.w));

      const motionState = new Ammo.btDefaultMotionState(transform);
      const rbInfo = new Ammo.btRigidBodyConstructionInfo(0, motionState, boxShape, new Ammo.btVector3(0, 0, 0));
      const body = new Ammo.btRigidBody(rbInfo);
      this.physics.world.addRigidBody(body);
    }
  }

  // -----------------------------------------------------------------
  //  Checkpoints – used for lap counting
  // -----------------------------------------------------------------
  placeCheckpoints() {
    const num = CONFIG.NUM_CHECKPOINTS;
    for (let i = 0; i < num; i++) {
      const t = i / num;
      const pos = this.spline.getPointAt(t);
      const checkpoint = {
        position: pos.clone(),
        radius: CONFIG.CHECKPOINT_RADIUS,
        index: i,
      };
      this.checkpoints.push(checkpoint);
    }
  }

  // -----------------------------------------------------------------
  //  Item Boxes – random placement along the track
  // -----------------------------------------------------------------
  placeItemBoxes() {
    const count = CONFIG.NUM_ITEM_BOXES;
    for (let i = 0; i < count; i++) {
      const t = Math.random();
      const pos = this.spline.getPointAt(t);
      const box = this.createItemBox(pos);
      this.itemBoxes.push(box);
    }
  }

  createItemBox(position) {
    const geometry = new THREE.BoxGeometry(1, 1, 1);
    const material = new THREE.MeshStandardMaterial({ color: 0xffd700 });
    const mesh = new THREE.Mesh(geometry, material);
    mesh.position.copy(position);
    mesh.userData.type = 'ItemBox';
    this.scene.add(mesh);
    return mesh;
  }

  // -----------------------------------------------------------------
  //  Helper – get nearest checkpoint index for a kart
  // -----------------------------------------------------------------
  getNearestCheckpoint(kartPos) {
    let best = { index: -1, dist: Infinity };
    for (const cp of this.checkpoints) {
      const d = kartPos.distanceTo(cp.position);
      if (d < best.dist) {
        best = { index: cp.index, dist: d };
      }
    }
    return best.index;
  }
}
```

### 4.4 ItemBox.js – Simple Item Logic

```js
// src/game/ItemBox.js
import * as THREE from 'three';
import { CONFIG } from '../config.js';

export default class ItemBox {
  constructor({ scene, position }) {
    this.scene = scene;
    this.position = position.clone();

    const geometry = new THREE.CylinderGeometry(0.5, 0.5, 0.3, 16);
    const material = new THREE.MeshStandardMaterial({ color: 0x00ff00 });
    this.mesh = new THREE.Mesh(geometry, material);
    this.mesh.position.copy(this.position);
    this.mesh.rotation.x = Math.PI / 2;
    this.mesh.userData.type = 'ItemBox';
    this.scene.add(this.mesh);
  }

  // Called when a kart collides with the box
  onPickup(kart) {
    // Randomly give one of three items
    const roll = Math.random();
    if (roll < 0.4) kart.startBoost(CONFIG.ITEM_BOOST_TIME, CONFIG.ITEM_BOOST_MULT);
    else if (roll < 0.7) kart.startBoost(CONFIG.ITEM_SHIELD_TIME, 1); // shield placeholder
    else kart.startBoost(CONFIG.ITEM_MUSHROOM_TIME, CONFIG.ITEM_MUSHROOM_MULT);
    // Remove visual
    this.scene.remove(this.mesh);
  }
}
```

```js
import SceneManager from '../core/SceneManager.js';
import InputManager from '../core/InputManager.js';
import Time from '../core/Time.js';
import Physics from '../core/Physics.js';
import Kart from './Kart.js';
import AIKart from './AIKart.js';
import Track from './Track.js';
import HUD from '../ui/HUD.js';
import { CONFIG } from '../config.js';

export default class GameManager {
  constructor() {
    // Core systems
    this.renderer = new Renderer({ canvas: document.querySelector('#glCanvas') });
    this.sceneMgr = new SceneManager();
    this.input = new InputManager();
    this.time = new Time();
    this.physics = new Physics();

    // Camera (follow‑cam)
    this.camera = new THREE.PerspectiveCamera(60, window.innerWidth / window.innerHeight, 0.1, 500);
    this.camera.position.set(0, 5, -10);
    this.camera.lookAt(0, 0, 0);
    this.renderer.setCamera(this.camera);

    // HUD
    this.hud = new HUD();

    // Build track
    const splinePoints = CONFIG.TRACK_SPLINE_POINTS.map(p => new THREE.Vector3(...p));
    this.track = new Track({
      scene: this.sceneMgr.getScene(),
      physics: this.physics,
      splinePoints,
    });

    // Player kart
    this.playerKart = new Kart({
      scene: this.sceneMgr.getScene(),
      physics: this.physics,
      input: this.input,
      modelUrl: '/assets/karts/player.glb',
      position: this.track.spline.getPointAt(0.01).add(new THREE.Vector3(0, 0.5, 0)),
    });

    // AI karts (2‑3 for demo)
    this.aiKarts = [];
    for (let i = 0; i < 3; i++) {
      const t = 0.05 + i * 0.02;
      const ai = new AIKart({
        scene: this.sceneMgr.getScene(),
        physics: this.physics,
        input: this.input, // not used by AI
        modelUrl: `/assets/karts/ai${i + 1}.glb`,
        position: this.track.spline.getPointAt(t).add(new THREE.Vector3(0, 0.5, 0)),
        path: this.track.spline,
      });
      this.aiKarts.push(ai);
    }

    // Lap / checkpoint tracking
    this.lap = 0;
    this.lastCheckpoint = -1;

    // Bind events
    this.sceneMgr.getScene().addEventListener('itemPickup', (e) => this.onItemPickup(e.kart));
    this.startLoop();
  }

  // -----------------------------------------------------------------
  //  Main loop – called via requestAnimationFrame
  // -----------------------------------------------------------------
  startLoop() {
    const loop = () => {
      this.time.tick();
      const dt = Math.min(this.time.delta, 1 / 30); // clamp for stability

      // Update physics first (so karts have fresh transforms)
      this.physics.step(dt);

      // Update player & AI
      this.playerKart.update(dt);
      this.aiKarts.forEach(ai => ai.update(dt));

      // Camera follow
      const kartPos = this.playerKart.mesh.position.clone();
      const camOffset = new THREE.Vector3(0, 5, -10).applyQuaternion(this.playerKart.mesh.quaternion);
      this.camera.position.copy(kartPos).add(camOffset);
      this.camera.lookAt(kartPos);

      // Check checkpoints / laps
      this.checkLapProgress(kartPos);

      // HUD update
      this.hud.update({
        speed: this.playerKart.speedKmh,
        lap: this.lap,
        position: this.getRacePosition(),
      });

      // Render
      this.renderer.render(this.sceneMgr.getScene());

      requestAnimationFrame(loop);
    };
    requestAnimationFrame(loop);
  }

  // -----------------------------------------------------------------
  //  Checkpoint / lap logic
  // -----------------------------------------------------------------
  checkLapProgress(kartPos) {
    const nearest = this.track.getNearestCheckpoint(kartPos);
    if (nearest === (this.lastCheckpoint + 1) % CONFIG.NUM_CHECKPOINTS) {
      this.lastCheckpoint = nearest;
      if (nearest === 0) {
        this.lap++;
        this.hud.showMessage(`Lap ${this.lap}`, 2);
      }
    }
  }

  // -----------------------------------------------------------------
  //  Simple race position (based on lap + checkpoint index)
  // -----------------------------------------------------------------
  getRacePosition() {
    const players = [this.playerKart, ...this.aiKarts];
    const scores = players.map(k => ({
      kart: k,
      score: this.lap * CONFIG.NUM_CHECKPOINTS + this.track.getNearestCheckpoint(k.mesh.position),
    }));
    scores.sort((a, b) => b.score - a.score);
    return scores.findIndex(s => s.kart === this.playerKart) + 1; // 1‑based
  }

  // -----------------------------------------------------------------
  //  Item pickup handling
  // -----------------------------------------------------------------
  onItemPickup(kart) {
    // Randomly give a boost for demo purposes
    kart.startBoost(CONFIG.ITEM_BOOST_TIME, CONFIG.ITEM_BOOST_MULT);
    this.hud.showMessage('Item! +Boost', 1);
  }
}
```

### 4.6 UI – HUD (src/ui)

**HUD.js**

```js
// src/ui/HUD.js
export default class HUD {
  constructor() {
    this.root = document.getElementById('hud');
    this.speedEl = this.root.querySelector('#speed');
    this.lapEl = this.root.querySelector('#lap');
    this.posEl = this.root.querySelector('#position');
    this.msgEl = this.root.querySelector('#msg');
  }

  update({ speed, lap, position }) {
    this.speedEl.textContent = `${Math.round(speed)} km/h`;
    this.lapEl.textContent = `Lap: ${lap}`;
    this.posEl.textContent = `Pos: ${position}`;
  }

  showMessage(text, seconds = 2) {
    this.msgEl.textContent = text;
    this.msgEl.style.opacity = 1;
    clearTimeout(this.msgTimeout);
    this.msgTimeout = setTimeout(() => {
      this.msgEl.style.opacity = 0;
    }, seconds * 1000);
  }
}
```

## 8️⃣ How to Continue From Here

1. **Replace placeholder assets** (`/assets/karts/*.glb`, `/assets/tracks/*`) with your own models/textures.  
2. **Fine‑tune physics** – adjust `KART_ACCEL`, `KART_STEER_ANGLE`, wheel friction, etc., until the handling feels “Mario‑Kart‑like”.  
3. **Add more items** (banana, shells, rockets) – each can be a subclass of `ItemBox` with its own effect.  
4. **Polish AI** – give each AI a distinct personality (aggressive, defensive).  
5. **Implement multiplayer** – start with a simple host‑client model using WebRTC; sync kart transforms and item usage.  
6. **Integrate post‑processing** – add Bloom, Motion Blur, and a subtle depth‑of‑field for that cinematic feel.  
7. **Run performance profiling** (Chrome DevTools → WebGL) and enable **instanced rendering** for track segments if you hit frame‑rate limits on mobile.

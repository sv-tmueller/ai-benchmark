function collide(a, b) {

function circCirc(a, b) {
  const dx = b.x - a.x, dy = b.y - a.y;
  const d = Math.hypot(dx, dy);
  const sr = a.r + b.r;
  if (d >= sr) return null;
  const nx = d > 0 ? dx / d : 1, ny = d > 0 ? dy / d : 0;
  return { normal: { x: nx, y: ny }, depth: sr - d, contact: { x: a.x + nx * a.r, y: a.y + ny * a.r } };
}
function circPoly(c, p) {
  const cs = Math.cos(-p.angle), sn = Math.sin(-p.angle);
  const lx = (c.x - p.x) * cs - (c.y - p.y) * sn;
  const ly = (c.x - p.x) * sn + (c.y - p.y) * cs;
  const hw = p.w / 2, hh = p.h / 2;
  const qx = Math.max(-hw, Math.min(hw, lx));
  const qy = Math.max(-hh, Math.min(hh, ly));
  const dx = lx - qx, dy = ly - qy;
  const dsq = dx * dx + dy * dy;
  if (dsq > c.r * c.r) return null;
  const d = Math.sqrt(dsq);
  let nlx, nly;
  if (d > 0.001) { nlx = dx / d; nly = dy / d; }
  else {
    const penR = hw - lx, penL = lx + hw, penT = hh - ly, penB = ly + hh;
    const mp = Math.min(penR, penL, penT, penB);
    if (mp === penR) { nlx = 1; nly = 0; }
    else if (mp === penL) { nlx = -1; nly = 0; }
    else if (mp === penT) { nlx = 0; nly = 1; }
    else { nlx = 0; nly = -1; }
  }
  const cw = Math.cos(p.angle), sw = Math.sin(p.angle);
  const wx = nlx * cw - nly * sw, wy = nlx * sw + nly * cw;
  return {
    normal: { x: -wx, y: -wy },
    depth: d > 0.001 ? c.r - d : c.r,
    contact: { x: c.x - wx * c.r, y: c.y - wy * c.r },
  };
}
function polyPoly(a, b) {
  const va = getVerts(a), vb = getVerts(b);
  const axes = [...edgeNormals(va), ...edgeNormals(vb)];
  let minO = Infinity, best = null;
  for (const ax of axes) {
    const pa = proj(va, ax), pb = proj(vb, ax);
    if (pa.max < pb.min || pb.max < pa.min) return null;
    const o = Math.min(pa.max, pb.max) - Math.max(pa.min, pb.min);
    if (o < minO) { minO = o; best = ax; }
  }
  const dx = b.x - a.x, dy = b.y - a.y;
  if (dx * best.x + dy * best.y < 0) best = { x: -best.x, y: -best.y };
  return { normal: best, depth: minO, contact: approxContact(va, vb, best) };
}
function edgeNormals(vs) {
  const out = [];
  for (let i = 0; i < vs.length; i++) {
    const v1 = vs[i], v2 = vs[(i + 1) % vs.length];
    const ex = v2.x - v1.x, ey = v2.y - v1.y;
    const l = Math.hypot(ex, ey) || 1;
    out.push({ x: -ey / l, y: ex / l });
  }
  return out;
}
function proj(vs, axis) {
  let mn = Infinity, mx = -Infinity;
  for (const v of vs) {
    const p = v.x * axis.x + v.y * axis.y;
    if (p < mn) mn = p; if (p > mx) mx = p;
  }
  return { min: mn, max: mx };
}
function approxContact(va, vb, axis) {
  let da = null, dm = Infinity;
  for (const v of va) { const p = -v.x * axis.x - v.y * axis.y; if (p < dm) { dm = p; da = v; } }
  let db = null, bm = Infinity;
  for (const v of vb) { const p = v.x * axis.x + v.y * axis.y; if (p < bm) { bm = p; db = v; } }
  return { x: (da.x + db.x) / 2, y: (da.y + db.y) / 2 };
}

// ------------------------- IMPULSE RESOLUTION ------------------------------
function resolve(a, b, m) {
  const n = m.normal, cp = m.contact;
  const rax = cp.x - a.x, ray = cp.y - a.y;
  const rbx = cp.x - b.x, rby = cp.y - b.y;
  const vax = a.vx - a.av * ray, vay = a.vy + a.av * rax;
  const vbx = b.vx - b.av * rby, vby = b.vy + b.av * rbx;
  const rvx = vbx - vax, rvy = vby - vay;
  const vn = rvx * n.x + rvy * n.y;
  if (vn > 0) return 0;
  const e = Math.min(a.rest, b.rest);
  const racn = rax * n.y - ray * n.x;
  const rbcn = rbx * n.y - rby * n.x;
  const ims = a.im + b.im + racn * racn * a.iI + rbcn * rbcn * b.iI;
  if (ims === 0) return 0;
  const j = -(1 + e) * vn / ims;
  const ix = j * n.x, iy = j * n.y;
  a.vx -= ix * a.im; a.vy -= iy * a.im; a.av -= racn * j * a.iI;
  b.vx += ix * b.im; b.vy += iy * b.im; b.av += rbcn * j * b.iI;
  // friction
  const tx = -n.y, ty = n.x;
  const vt = rvx * tx + rvy * ty;
  let jt = -vt / ims;
  const mu = Math.sqrt(a.fric * b.fric);
  if (Math.abs(jt) > Math.abs(j) * mu) jt = Math.abs(j) * mu * Math.sign(jt);
  a.vx -= tx * jt * a.im; a.vy -= ty * jt * a.im; a.av -= (rax * ty - ray * tx) * jt * a.iI;
  b.vx += tx * jt * b.im; b.vy += ty * jt * b.im; b.av += (rbx * ty - rby * tx) * jt * b.iI;
  return Math.abs(vn);
}
function positionalCorrect(a, b, m) {
  const pct = 0.55, slop = 0.04;
  const ims = a.im + b.im;
  if (ims === 0) return;
  const corr = Math.max(m.depth - slop, 0) / ims * pct;
  a.x -= corr * m.normal.x * a.im;
  a.y -= corr * m.normal.y * a.im;
  b.x += corr * m.normal.x * b.im;
  b.y += corr * m.normal.y * b.im;
}

// -------------------------- PHYSICS STEP -----------------------------------
function physicsStep(dt) {
  for (const b of bodies) {
    if (!b.alive || b.sleeping) continue;
    b.vy += GRAVITY * dt;
    b.vx *= 0.998; b.vy *= 0.998; b.av *= 0.985;
    const vm = Math.hypot(b.vx, b.vy);
    if (vm > 2200) { b.vx *= 2200 / vm; b.vy *= 2200 / vm; }
    if (Math.abs(b.av) > 30) b.av = 30 * Math.sign(b.av);
    b.x += b.vx * dt; b.y += b.vy * dt; b.angle += b.av * dt;
  }
  for (let i = 0; i < bodies.length; i++) {
    const a = bodies[i]; if (!a.alive) continue;
    for (let j = i + 1; j < bodies.length; j++) {
      const b = bodies[j]; if (!b.alive) continue;
      if (a.sleeping && b.sleeping) continue;
      const rA = a.shape === "circle" ? a.r : Math.hypot(a.w, a.h) / 2;
      const rB = b.shape === "circle" ? b.r : Math.hypot(b.w, b.h) / 2;
      if (Math.hypot(a.x - b.x, a.y - b.y) > rA + rB + 6) continue;
      const m = collide(a, b);
      if (m) {
        a.sleeping = false; b.sleeping = false;
        a.sleepTimer = 0; b.sleepTimer = 0;
        const imp = resolve(a, b, m);
        positionalCorrect(a, b, m);
        if (imp > 170) {
          spawnDust(m.contact.x, m.contact.y, Math.min(10, imp * 0.03));
          if (a.kind === "pig") damagePig(a, imp);
          if (b.kind === "pig") damagePig(b, imp);
          if (a.kind === "box") damageBox(a, imp);
          if (b.kind === "box") damageBox(b, imp);
          if (imp > 380) addShake(Math.min(8, imp * 0.012));
        }
      }
    }
  }
  for (const b of bodies) if (b.alive) collideGround(b);
  for (const b of bodies) {
    if (!b.alive) continue;
    if (b.x < -120 || b.x > W + 120 || b.y > H + 120) b.alive = false;
  }
  for (const b of bodies) {
    if (!b.alive || b.kind === "bird") continue;
    const sq = b.vx * b.vx + b.vy * b.vy + b.av * b.av * 40;
    if (sq < 8) {
      b.sleepTimer += dt;
      if (b.sleepTimer > 0.6) { b.sleeping = true; b.vx = 0; b.vy = 0; b.av = 0; }
    } else { b.sleepTimer = 0; b.sleeping = false; }
  }
}

function launchBird() {
  if (!activeBird || !activeBird.onSling) return;
  const dx = SLING.x - dragPos.x, dy = SLING.y - dragPos.y;
  const mag = Math.hypot(dx, dy);
  if (mag < 12) {
    dragging = false;
    activeBird.x = SLING.x; activeBird.y = SLING.y;
    dragPos = { x: SLING.x, y: SLING.y };
    return;
  }
  const clamped = Math.min(mag, MAX_DRAG);
  const ux = dx / mag, uy = dy / mag;
  const speed = clamped * LAUNCH_SCALE;
  activeBird.vx = ux * speed; activeBird.vy = uy * speed;
  activeBird.x = dragPos.x; activeBird.y = dragPos.y;
  activeBird.onSling = false; activeBird.launched = true;
  bodies.push(activeBird);
  birdsRemaining--; birdsUsed++;
  dragging = false;
  addShake(3);
}
function retireActiveBird() {
  if (!activeBird) return;
  activeBird.alive = false;
  activeBird = null;
  if (birdsRemaining > 0) pendingReload = 0.45;
}
function checkProgress() {
  if (state !== "playing") return;
  const pigs = bodies.filter(b => b.kind === "pig" && b.alive);
  if (pigs.length === 0) { finishLevel(true); return; }
  if (birdsRemaining <= 0 && !activeBird && pendingReload <= 0) finishLevel(false);
}
function finishLevel(won) {
  if (won) {
    const stars = birdsUsed === 1 ? 3 : birdsUsed === 2 ? 2 : 1;
    lastStars = stars; totalStars += stars;
    score += birdsRemaining * 10000;
    state = currentLevel < LEVELS.length - 1 ? "levelComplete" : "allWon";
  } else state = "lost";
}

// ------------------------------- UPDATE ------------------------------------
function update(dt) {
  if (state !== "playing") return;
  if (pendingReload > 0) {
    pendingReload -= dt;
    if (pendingReload <= 0 && !activeBird && birdsRemaining > 0) loadBirdOnSling();
  }
  const sd = dt / SUBSTEPS;
  for (let i = 0; i < SUBSTEPS; i++) physicsStep(sd);
  updateParticles(dt);
  if (activeBird && activeBird.launched) {
    activeBird.timeAlive += dt;
    const sp = Math.hypot(activeBird.vx, activeBird.vy);
    if (sp < 25 && activeBird.sleeping) {
      activeBird.settleTimer += dt;
      if (activeBird.settleTimer > 1.0) retireActiveBird();
    } else activeBird.settleTimer = 0;
    if (activeBird && (activeBird.x < -60 || activeBird.x > W + 60 || activeBird.y > H + 60)) retireActiveBird();
    if (activeBird && activeBird.timeAlive > 10) retireActiveBird();
  }
  if (Math.random() < 0.04) bodies = bodies.filter(b => b.alive);
  checkProgress();
  shake *= 0.86;
}

// ------------------------------- RENDER ------------------------------------
function render() {
  ctx.clearRect(0, 0, W, H);
  drawBackground();
  ctx.save();
  if (shake > 0.1) ctx.translate((Math.random() - 0.5) * shake, (Math.random() - 0.5) * shake);
  drawGround();
  drawSlingBack();
  drawBodies();
  if (activeBird && activeBird.onSling) drawBirdOnSling();
  drawSlingFront();
  if (dragging && activeBird && activeBird.onSling) drawTrajectoryPreview();
  if (activeBird && activeBird.onSling && !dragging) drawPullHint();
  drawParticles();
  ctx.restore();
  drawHUD();
  drawOverlay();
}

function drawTrajectoryPreview() {
  const dx = SLING.x - dragPos.x, dy = SLING.y - dragPos.y;
  const mag = Math.hypot(dx, dy);
  if (mag < 12) return;
  const clamped = Math.min(mag, MAX_DRAG);
  const ux = dx / mag, uy = dy / mag;
  const speed = clamped * LAUNCH_SCALE;
  let x = dragPos.x, y = dragPos.y, vx = ux * speed, vy = uy * speed;
  const dt = 0.04;
  for (let i = 0; i < 42; i++) {
    vy += GRAVITY * dt;
    x += vx * dt; y += vy * dt;
    if (y > GROUND_Y || x > W || x < 0) break;
    const sz = Math.max(1.5, 4 - i * 0.07);
    ctx.globalAlpha = Math.max(0.2, 1 - i * 0.022);
    ctx.fillStyle = "#fff";
    ctx.beginPath(); ctx.arc(x, y, sz, 0, Math.PI * 2); ctx.fill();
  }
  ctx.globalAlpha = 1;
}

// -------------------------------- INPUT ------------------------------------
function getCanvasPos(clientX, clientY) {
  const r = canvas.getBoundingClientRect();
  return { x: (clientX - r.left) * (canvas.width / r.width), y: (clientY - r.top) * (canvas.height / r.height) };
}
function updateDrag(mx, my) {
  let x = mx, y = my;
  const dx = x - SLING.x, dy = y - SLING.y;
  const mag = Math.hypot(dx, dy);
  if (mag > MAX_DRAG) { x = SLING.x + dx / mag * MAX_DRAG; y = SLING.y + dy / mag * MAX_DRAG; }
  if (x > SLING.x - 5) x = SLING.x - 5;
  dragPos = { x, y };
  if (activeBird) { activeBird.x = x; activeBird.y = y; }
}
function pointerDown(mx, my) {
  if (state === "levelComplete") { startLevel(currentLevel + 1); return; }
  if (state === "allWon") { currentLevel = 0; score = 0; totalStars = 0; lastStars = 0; startLevel(0); return; }
  if (state === "lost") { startLevel(currentLevel); return; }
  if (state !== "playing" || !activeBird || !activeBird.onSling) return;
  if (Math.hypot(mx - activeBird.x, my - activeBird.y) < 55) {
    dragging = true;
    updateDrag(mx, my);
  }
}
function pointerMove(mx, my) { if (dragging) updateDrag(mx, my); }
function pointerUp() { if (dragging) launchBird(); }

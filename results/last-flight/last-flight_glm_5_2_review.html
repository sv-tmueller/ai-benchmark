```typescript
export function safeNormalize(v: THREE.Vector3): THREE.Vector3 {
  const lenSq = v.lengthSq();
  if (lenSq < 1e-8) return v.set(0, 0, 0);
  return v.multiplyScalar(1 / Math.sqrt(lenSq));
}

export function damp(current: number, target: number, lambda: number, dt: number): number {
  return THREE.MathUtils.lerp(current, target, 1 - Math.exp(-lambda * dt));
}

export function dampVec(out: THREE.Vector3, current: THREE.Vector3, target: THREE.Vector3, lambda: number, dt: number): void {
  out.x = damp(current.x, target.x, lambda, dt);
  out.y = damp(current.y, target.y, lambda, dt);
  out.z = damp(current.z, target.z, lambda, dt);
}
```

```typescript
export type EventType = 
  | 'jetKilled' | 'caught' | 'tailTorn' | 'touchdown' 
  | 'phaseChanged' | 'heroHit' | 'hullHit' | 'boom';

interface EventPayload {
  [key: string]: any;
}

type Listener<T extends EventPayload> = (payload: T) => void;

class EventBus {
  private listeners: Map<EventType, Listener<any>[]> = new Map();

  on<T extends EventPayload>(event: EventType, cb: Listener<T>): () => void {
    if (!this.listeners.has(event)) this.listeners.set(event, []);
    this.listeners.get(event)!.push(cb);
    return () => this.off(event, cb);
  }

  off<T extends EventPayload>(event: EventType, cb: Listener<T>): void {
    const arr = this.listeners.get(event);
    if (arr) {
      const i = arr.indexOf(cb);
      if (i !== -1) arr.splice(i, 1);
    }
  }

  emit<T extends EventPayload>(event: EventType, payload?: T): void {
    const arr = this.listeners.get(event);
    if (arr) arr.forEach(cb => cb(payload || {}));
  }
}

export const Events = new EventBus();
```

### src/core/pool.ts
```typescript
export class Pool<T> {
  private items: T[] = [];
  private factory: () => T;
  private resetFn: (item: T) => void;

  constructor(factory: () => T, resetFn: (item: T) => void, prealloc: number = 0) {
    this.factory = factory;
    this.resetFn = resetFn;
    for (let i = 0; i < prealloc; i++) this.items.push(factory());
  }

  acquire(): T {
    if (this.items.length > 0) return this.items.pop()!;
    return this.factory();
  }

  Release(item: T): void {
    this.resetFn(item);
    this.items.push(item);
  }

  ReleaseAll(activeItems: T[]): void {
    activeItems.forEach(it => this.Release(it));
    activeItems.length = 0;
  }
}
```

```typescript
export interface InputState {
  mouseX: number;
  mouseY: number;
  keys: Record<string, boolean>;
  mouseButtons: number[];
  wheelDelta: number;
  locked: boolean;
}

export function createInput(target: HTMLElement): { state: InputState, update: () => void, dispose: () => void } {
  const state: InputState = {
    mouseX: 0, mouseY: 0, keys: {}, mouseButtons: [], wheelDelta: 0, locked: false,
  };

  const onKeyDown = (e: KeyboardEvent) => { state.keys[e.code] = true; };
  const onKeyUp = (e: KeyboardEvent) => { state.keys[e.code] = false; };
  const onMouseMove = (e: MouseEvent) => {
    if (state.locked) {
      state.mouseX += e.movementX;
      state.mouseY += e.movementY;
    } else {
      state.mouseX = e.clientX;
      state.mouseY = e.clientY;
    }
  };
  const onMouseDown = (e: MouseEvent) => { if(!state.mouseButtons.includes(e.button)) state.mouseButtons.push(e.button); };
  const onMouseUp = (e: MouseEvent) => { state.mouseButtons = state.mouseButtons.filter(b => b !== e.button); };
  const onWheel = (e: WheelEvent) => { state.wheelDelta += e.deltaY; };
  const onPointerChange = () => { state.locked = document.pointerLockElement === target; };

  target.addEventListener('keydown', onKeyDown);
  target.addEventListener('keyup', onKeyUp);
  target.addEventListener('mousemove', onMouseMove);
  target.addEventListener('mousedown', onMouseDown);
  target.addEventListener('mouseup', onMouseUp);
  target.addEventListener('wheel', onWheel);
  document.addEventListener('pointerlockchange', onPointerChange);

  const update = () => {
    state.wheelDelta = 0;
  };

  const dispose = () => {
    target.removeEventListener('keydown', onKeyDown);
    target.removeEventListener('keyup', onKeyUp);
    target.removeEventListener('mousemove', onMouseMove);
    target.removeEventListener('mousedown', onMouseDown);
    target.removeEventListener('mouseup', onMouseUp);
    target.removeEventListener('wheel', onWheel);
    document.removeEventListener('pointerlockchange', onPointerChange);
  };

  return { state, update, dispose };
}
```

```typescript
export function createAudio(ctx: AudioContext) {
  const masterGain = ctx.createGain();
  masterGain.gain.value = 0.6;
  masterGain.connect(ctx.destination);

  function playBoom(intensity: number = 1) {
    const t = ctx.currentTime;
    const osc = ctx.createOscillator();
    const g = ctx.createGain();
    osc.type = 'sine';
    osc.frequency.setValueAtTime(120, t);
    osc.frequency.exponentialRampToValueAtTime(30, t + 0.4);
    g.gain.setValueAtTime(0.8 * intensity, t);
    g.gain.exponentialRampToValueAtTime(0.001, t + 0.4);
    osc.connect(g).connect(masterGain);
    osc.start(t);
    osc.stop(t + 0.4);
    
    const bufferSize = ctx.sampleRate * 0.4;
    const buffer = ctx.createBuffer(1, bufferSize, ctx.sampleRate);
    const data = buffer.getChannelData(0);
    for (let i = 0; i < bufferSize; i++) data[i] = (Math.random() * 2 - 1) * (1 - i / bufferSize);
    const noise = ctx.createBufferSource();
    noise.buffer = buffer;
    const lp = ctx.createBiquadFilter();
    lp.type = 'lowpass';
    lp.frequency.value = 800;
    noise.connect(lp).connect(masterGain);
    noise.start(t);
  }

  function playLaser() {
    const t = ctx.currentTime;
    const osc = ctx.createOscillator();
    const g = ctx.createGain();
    osc.type = 'sawtooth';
    osc.frequency.value = 800;
    g.gain.setValueAtTime(0.1, t);
    g.gain.exponentialRampToValueAtTime(0.001, t + 0.2);
    osc.connect(g).connect(masterGain);
    osc.start(t);
    osc.stop(t + 0.2);
  }

  return { masterGain, playBoom, playLaser };
}
```

```typescript
import { CONFIG } from '../config';

export function createRenderer(container: HTMLElement) {
  const renderer = new THREE.WebGLRenderer({ antialias: true, powerPreference: 'high-performance' });
  renderer.setSize(window.innerWidth, window.innerHeight);
  renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
  renderer.shadowMap.enabled = true;
  renderer.shadowMap.type = THREE.PCFSoftShadowMap;
  renderer.toneMapping = THREE.NoToneMapping; // Handled by post
  container.appendChild(renderer.domElement);

  const scene = new THREE.Scene();
  const camera = new THREE.PerspectiveCamera(CONFIG.Camera.FovBase, window.innerWidth / window.innerHeight, 0.1, 20000);

  let composer: EffectComposer | null = null;
  try {
    composer = new EffectComposer(renderer, { multisampling: 4, frameBufferType: THREE.HalfFloatType });
    composer.addPass(new RenderPass(scene, camera));
    const effects = [
      new BloomEffect({ mipmapBlur: true, luminanceThreshold: 0.8, intensity: 0.9, radius: 0.6 }),
      new SMAAEffect(),
      new ToneMappingEffect({ mode: ToneMappingMode.ACES_FILMIC })
    ];
    composer.addPass(new EffectPass(camera, ...effects));
  } catch (e) {
    console.error("Post processing failed, fallback to raw render", e);
    composer = null;
  }

  const resize = () => {
    const w = window.innerWidth, h = window.innerHeight;
    renderer.setSize(w, h);
    camera.aspect = w / h;
    camera.updateProjectionMatrix();
    if(composer) composer.setSize(w, h);
  };
  window.addEventListener('resize', resize);

  const render = (dt: number) => {
    if (composer) composer.render(dt);
    else renderer.render(scene, camera);
  };

  return { renderer, scene, camera, render, resize: () => {} /*handled*/ };
}
```

```typescript
import { CONFIG } from '../config';

export function createLoop(update: (fixedStep: number, frameDt: number, timeScale: number) => void) {
  let rafId = 0;
  let lastTime = performance.now();
  let acc = 0;
  let running = false;
  let timeScale = 1.0;

  const tick = () => {
    if (!running) return;
    rafId = requestAnimationFrame(tick);
    
    const now = performance.now();
    let frameDt = (now - lastTime) / 1000;
    lastTime = now;
    
    if (frameDt > CONFIG.Loop.MaxFrameDt) frameDt = CONFIG.Loop.MaxFrameDt;
    
    acc += frameDt * timeScale;
    const fixedStep = CONFIG.Loop.FixedDt;
    
    let steps = 0;
    while (acc >= fixedStep && steps < 10) {
      update(fixedStep, 0, timeScale);
      acc -= fixedStep;
      steps++;
    }
    
    update(0, frameDt, timeScale); // Frame update for interpolation/render prep
  };

  const start = () => {
    if(running) return;
    running = true;
    lastTime = performance.now();
    rafId = requestAnimationFrame(tick);
  };

  const stop = () => {
    running = false;
    cancelAnimationFrame(rafId);
  };

  const setTimeScale = (ts: number) => { timeScale = ts; };

  return { start, stop, setTimeScale };
}
```

```typescript
import * as THREE from 'three';
import { CONFIG } from '../config';
import { safeNormalize, damp } from '../core/math';

export interface FlightCtx {
  bounds: THREE.Box3;
}

export interface FlightInput {
  thrust: number; // 0..1
  boost: boolean;
  brake: boolean;
  up: number; // -1..1
  strafe: number; // -1..1
  aimYaw: number;
  aimPitch: number;
}

export enum FlightMode { Hover, Flight, Holding, Landed, Stunned }

export class FlightModel {
  pos = new THREE.Vector3(0, 100, 0);
  vel = new THREE.Vector3();
  quat = new THREE.Quaternion();
  aimYaw = 0;
  aimPitch = 0;
  boostCharge = 1;
  heat = 0;
  hp = CONFIG.Hero.Hp;
  mode = FlightMode.Hover;
  stunTimer = 0;
  regenTimer = 0;
  bank = 0;
  pitchVis = 0;
  sonicArmed = true;
  hasBoomed = false;

  private tmpAim = new THREE.Vector3();
  private tmpVelDir = new THREE.Vector3();
  private fwd = new THREE.Vector3(0, 0, -1);
  private up = new THREE.Vector3(0, 1, 0);

  update(dt: number, input: FlightInput, ctx: FlightCtx) {
    if (this.mode === FlightMode.Stunned) {
      this.stunTimer -= dt;
      this.vel.y -= CONFIG.Hero.StunGravity * dt;
      this.pos.addScaledVector(this.vel, dt);
      
      // Simple ground collision for stun
      if (this.pos.y <= 1.8) {
        this.pos.y = 1.8;
        this.vel.y *= -0.25;
        this.vel.x *= 0.4; this.vel.z *= 0.4;
        if (this.vel.lengthSq() < 1) this.mode = FlightMode.Landed;
      }
      
      if (this.stunTimer <= 0) this.mode = FlightMode.Hover;
      return;
    }

    // Update aim
    this.aimYaw = input.aimYaw;
    this.aimPitch = input.aimPitch;
    
    // Calculate aim vector
    const cy = Math.cos(this.aimYaw), sy = Math.sin(this.aimYaw);
    const cp = Math.cos(this.aimPitch), sp = Math.sin(this.aimPitch);
    this.tmpAim.set(-sy * cp, sp, -cy * cp);
    
    const speed = this.vel.length();
    
    // Turn rate
    const speedNorm = Math.min(speed / CONFIG.Hero.BoostSpeed, 1);
    const turnRate = THREE.MathUtils.lerp(CONFIG.Hero.TurnRateMax, CONFIG.Hero.TurnRateMin, THREE.MathUtils.smoothstep(speedNorm, 0, 1));
    
    // Rotate velocity towards aim
    if (speed > 0.1) {
      this.tmpVelDir.copy(this.vel).normalize();
      const qCurr = new THREE.Quaternion().setFromUnitVectors(this.fwd, this.tmpVelDir);
      const qTarget = new THREE.Quaternion().setFromUnitVectors(this.fwd, this.tmpAim);
      qCurr.rotateTowards(qTarget, turnRate * dt);
      this.tmpVelDir.set(0,0,-1).applyQuaternion(qCurr);
      this.vel.copy(this.tmpVelDir).multiplyScalar(speed);
    } else {
      this.vel.copy(this.tmpAim).multiplyScalar(0.01); // micro nudge to align
    }

    // Boost management
    if (input.boost && this.boostCharge > 0 && !input.brake) {
      this.boostCharge -= dt / CONFIG.Hero.BoostDrainTime;
      if (this.boostCharge < 0) this.boostCharge = 0;
    } else {
      this.boostCharge += dt / CONFIG.Hero.BoostRefillTime;
      if (this.boostCharge > 1) this.boostCharge = 1;
    }

    // Forces
    const isBoosting = input.boost && this.boostCharge > 0;
    const thrustMag = input.thrust * CONFIG.Hero.Thrust * (isBoosting ? CONFIG.Hero.BoostMult : 1);
    const vMax = isBoosting ? CONFIG.Hero.BoostSpeed : CONFIG.Hero.CruiseSpeed;
    const dragK = isBoosting ? CONFIG.Hero.DragBoost : CONFIG.Hero.DragCruise;
    
    // Apply thrust
    if (thrustMag > 0) {
      this.vel.addScaledVector(this.tmpAim, thrustMag * dt);
    }
    
    // Vertical & Strafe
    this.vel.y += input.up * CONFIG.Hero.VertAccel * dt;
    this.vel.addScaledVector(new THREE.Vector3(cy, 0, -sy), input.strafe * CONFIG.Hero.StrafeAccel * dt);
    
    // Gravity
    if (thrustMag === 0 && input.up === 0) {
      this.vel.y -= CONFIG.Hero.CoastGravity * dt;
    }
    
    // Drag
    const dragF = this.vel.clone().multiplyScalar(-this.vel.length() * dragK);
    this.vel.addScaledVector(dragF, dt);
    
    // Clamp speed
    if (this.vel.length() > vMax) this.vel.normalize().multiplyScalar(vMax);
    
    // Integrate
    this.pos.addScaledVector(this.vel, dt);
    
    // Bounds check
    if (this.pos.distanceTo(new THREE.Vector3()) > 4500) {
      const pushBack = this.pos.clone().normalize().negate().multiplyScalar(10 * dt);
      this.vel.add(pushBack);
    }
    
    // Sonic Boom
    if (speed >= 140 && this.sonicArmed) {
      this.hasBoomed = true;
      this.sonicArmed = false;
    }
    if (speed < 120) this.sonicArmed = true;

    // Body Orientation
    this.updateOrientation(dt, input, speed, isBoosting);
    
    // HP Regen
    if (this.regenTimer > 0) this.regenTimer -= dt;
    else if (this.hp < CONFIG.Hero.Hp) this.hp = Math.min(CONFIG.Hero.Hp, this.hp + CONFIG.Hero.RegenRate * dt);
  }

  private updateOrientation(dt: number, input: FlightInput, speed: number, isBoosting: boolean) {
    const targetMode = speed < 6 ? FlightMode.Hover : FlightMode.Flight;
    if (this.mode !== FlightMode.Holding && this.mode !== FlightMode.Landed) this.mode = targetMode;
    
    const targetQ = new THREE.Quaternion().setFromUnitVectors(this.fwd, this.tmpAim);
    const slerpRate = isBoosting ? 10 : 6;
    this.quat.slerp(targetQ, 1 - Math.exp(-slerpRate * dt));
    
    // Banking
    const yawRate = (this.aimYaw - (this.prevYaw ?? this.aimYaw)) / dt;
    this.prevYaw = this.aimYaw;
    const speedFactor = Math.min(speed / CONFIG.Hero.CruiseSpeed, 1);
    const targetBank = THREE.MathUtils.clamp(-yawRate * speedFactor * 1.4, -70, 70) * Math.PI / 180;
    this.bank = damp(this.bank, targetBank, 1 / 0.15, dt);
    
    const targetPitch = this.aimPitch + (input.brake ? 15 : 0);
    this.pitchVis = damp(this.pitchVis, targetPitch, 1 / 0.15, dt);
    
    // Apply bank visually
    const finalQ = new THREE.Quaternion().setFromEuler(new THREE.Euler(this.pitchVis*Math.PI/180, this.aimYaw, this.bank));
    this.quat.copy(finalQ);
  }

  prevYaw?: number;

  takeDamage(amount: number) {
    this.hp -= amount;
    this.regenTimer = CONFIG.Hero.RegenDelay;
    if (this.hp <= 0) {
      this.hp = 0;
      this.mode = FlightMode.Stunned;
      this.stunTimer = 2.0;
```

```typescript
import * as THREE from 'three';
import { CONFIG } from '../config';
import { safeNormalize } from '../core/math';

export class HoldingController {
  vel = new THREE.Vector3(); // Plane velocity
  pos = new THREE.Vector3(); // Plane pos
  stress = 0;
  hullHp = CONFIG.Airliner.Hp;
  brakeAmount = 0;
  yawRate = 0;
  pitch = 0;
  roll = 0;
  prevYaw = 0;
  
  init(initPos: THREE.Vector3, initVel: THREE.Vector3) {
    this.pos.copy(initPos);
    this.vel.copy(initVel);
    this.stress = 0;
    this.hullHp = CONFIG.Airliner.Hp;
    this.brakeAmount = 0;
  }
  
  update(dt: number, input: {brake: boolean, forward: boolean, up: number, down: boolean, aimYaw: number, aimPitch: number}) {
    // Ramp brake
    const targetBrake = input.brake ? 1 : 0;
    this.brakeAmount = THREE.MathUtils.damp(this.brakeAmount, targetBrake, 1 / CONFIG.Holding.BrakeRampTime, dt);
    const brakeForce = this.brakeAmount * CONFIG.Holding.BrakeMax;
    
    // Forward/back
    const fAccel = input.forward ? CONFIG.Holding.PushForward : 0;
    const bAccel = -brakeForce;
    
    // Gravity & Lift
    let yAccel = -CONFIG.Holding.Gravity;
    if(input.up > 0) yAccel += CONFIG.Holding.Lift;
    if(input.down) yAccel -= CONFIG.Holding.DownForce;
    
    // Attitude (lagged)
    const targetPitch = THREE.MathUtils.clamp(input.aimPitch, -CONFIG.Holding.PitchLimit, CONFIG.Holding.PitchLimit);
    const targetRoll = THREE.MathUtils.clamp((this.yawRate * 10), -CONFIG.Holding.RollLimit, CONFIG.Holding.RollLimit);
    this.pitch = THREE.MathUtils.damp(this.pitch, targetPitch, 2, dt);
    this.roll = THREE.MathUtils.damp(this.roll, targetRoll, 2, dt);
    
    const yawDiff = input.aimYaw - this.prevYaw;
    this.prevYaw = input.aimYaw;
    this.yawRate = THREE.MathUtils.clamp(yawDiff / dt, -CONFIG.Holding.YawRate, CONFIG.Holding.YawRate);
    
    // Aerodynamics
    if(this.pitch > CONFIG.Holding.AeroBrakePitch) {
      yAccel += CONFIG.Holding.AeroLiftForce;
      this.vel.multiplyScalar(1 - (CONFIG.Holding.AeroBrakeForce * dt / 100)); // mild drag
    } else if(this.pitch < -CONFIG.Holding.NoseDownPitch) {
      yAccel -= CONFIG.Holding.NoseDownForce;
    }
    
    // Apply accelerations
    const fwd = new THREE.Vector3(-Math.sin(input.aimYaw), 0, -Math.cos(input.aimYaw)).multiplyScalar(fAccel + bAccel);
    this.vel.x += fwd.x * dt;
    this.vel.z += fwd.z * dt;
    this.vel.y += yAccel * dt;
    
    // Drag (simple linear to simulate air resistance without exact terminal v)
    this.vel.multiplyScalar(1 - 0.5 * dt);
    
    // Integrate
    this.pos.addScaledVector(this.vel, dt);
    
    // Stress calculation
    let stressRise = 0;
    if(brakeForce > 7) stressRise += CONFIG.Holding.StressRiseRate * ((brakeForce - 7)/5);
    if(Math.abs(this.yawRate) > 0.3) stressRise += CONFIG.Holding.StressTurnRate;
    if(this.vel.length() > 130) stressRise += CONFIG.Holding.StressTurnRate;
    if(this.vel.y < -20) stressRise += CONFIG.Holding.StressTurnRate;
    
    if(stressRise > 0) this.stress += stressRise * dt;
    else this.stress -= CONFIG.Holding.StressDecayRate * dt;
    
    this.stress = THREE.MathUtils.clamp(this.stress, 0, 100);
    
    if(this.stress >= 100) {
      this.hullHp -= CONFIG.Holding.StressHullDPS * dt;
    }
  }
  
  applyDamage(dmg: number) {
    this.hullHp -= dmg;
  }
}
```

```typescript
import * as THREE from 'three';
import { CONFIG } from '../config';

export enum Phase { Title, Cinematic, Intercept, Catch, Arrest, SetDown, Results, Failed }
export type FailReason = 'impact' | 'hull_zero' | 'hero_death' | 'crash_touchdown' | '';

export class MissionManager {
  phase = Phase.Title;
  failReason: FailReason = '';
  checkpoint = Phase.Intercept;
  startTime = 0;
  
  init() {
    this.startTime = performance.now();
    this.phase = Phase.Title;
  }
  
  startGame() {
    this.phase = Phase.Cinematic;
    this.checkpoint = Phase.Intercept;
  }
  
  update(heroPos: THREE.Vector3, planePos: THREE.Vector3, planeHp: number, heroHp: number) {
    if (this.phase === Phase.Failed || this.phase === Phase.Results) return;
    
    const dist = heroPos.distanceTo(planePos);
    
    if (this.phase === Phase.Intercept && dist < 1500) {
      this.phase = Phase.Catch;
      this.checkpoint = Phase.Catch;
    }
    
    if (this.phase === Phase.Catch && this.isCaught) {
      this.phase = Phase.Arrest;
      this.checkpoint = Phase.Arrest;
    }
    
    if (planeHp <= 0) this.fail('hull_zero');
    if (heroHp <= 0) this.fail('hero_death');
  }
  
  isCaught = false;
  setIsCaught(v: boolean) { this.isCaught = v; }
  
  enterSetDown() {
    if (this.phase === Phase.Arrest) {
      this.phase = Phase.SetDown;
      this.checkpoint = Phase.SetDown;
    }
  }
  
  succeedTouchdown() {
    this.phase = Phase.Results;
  }
  
  fail(reason: Exclude<FailReason, ''>) {
    this.phase = Phase.Failed;
    this.failReason = reason;
  }
  
  retry() {
    this.phase = this.checkpoint;
    this.failReason = '';

  }
```

### src/game/camera.ts
```typescript
import * as THREE from 'three';
import { CONFIG } from '../config';
import { FlightModel } from '../entities/flight';
import { dampVec } from '../core/math';

export function createCamera(camera: THREE.PerspectiveCamera) {
  const target = new THREE.Vector3();
  const desiredPos = new THREE.Vector3();
  
  const update = (dt: number, hero: FlightModel, aimDir: THREE.Vector3, speed: number) => {
    const speedNorm = Math.min(speed / CONFIG.Hero.BoostSpeed, 1);
    const dist = THREE.MathUtils.lerp(CONFIG.Camera.DistHover, CONFIG.Camera.DistSupersonic, speedNorm);
    const height = THREE.MathUtils.lerp(CONFIG.Camera.HeightHover, CONFIG.Camera.HeightSupersonic, speedNorm);
    
    desiredPos.copy(hero.pos).addScaledVector(aimDir, -dist);
    desiredPos.y += height;
    
    const lag = speedNorm > 0.7 ? CONFIG.Camera.LagBoost : CONFIG.Camera.LagCruise;
    dampVec(camera.position, camera.position, desiredPos, 1/lag, dt);
    
    const lookAhead = THREE.MathUtils.lerp(CONFIG.Camera.LookAheadMin, CONFIG.Camera.LookAheadMax, speedNorm);
    target.copy(hero.pos).addScaledVector(aimDir, lookAhead);
    camera.lookAt(target);
    
    const fov = CONFIG.Camera.FovBase + 30 * THREE.MathUtils.smoothstep(speedNorm, 0, 1);
    camera.fov = damp(camera.fov, fov, 8, dt);
    camera.updateProjectionMatrix();
  };
  
  return { update };
}

function damp(a:number,b:number,c:number,d:number){return THREE.MathUtils.lerp(a,b,1-Math.exp(-c*d));}
```

```typescript
import * as THREE from 'three';
import { createRenderer } from './core/renderer';
import { createLoop } from './core/loop';
import { createInput } from './core/input';
import { createAudio } from './core/audio';
import { CONFIG } from './config';
import { FlightModel, FlightInput } from './entities/flight';
import { createHero } from './entities/hero';
import { createCity } from './world/city';
import { createGround } from './world/ground';
import { createSky } from './world/sky';
import { createCamera } from './game/camera';
import { createHUD } from './game/hud';
import { createDebugGUI } from './game/debug';
import { safeNormalize } from './core/math';

const container = document.getElementById('app')!;

const { renderer, scene, camera, render } = createRenderer(container);
const inputMgr = createInput(document.body);
const audioCtx = new (window.AudioContext || (window as any).webkitAudioContext)();
const audio = createAudio(audioCtx);

const ambient = new THREE.AmbientLight(0x404040);
scene.add(ambient);
const dir = new THREE.DirectionalLight(0xffffff, 1.5);
dir.position.set(100, 200, 100);
dir.castShadow = true;
dir.shadow.mapSize.width = 2048;
dir.shadow.mapSize.height = 2048;
scene.add(dir);

createSky(scene);
createGround(scene);
createCity(scene);

const heroEntity = createHero(scene);
const flight = new FlightModel();

const camCtrl = createCamera(camera);
const hud = createHUD(container);
createDebugGUI();

container.addEventListener('click', () => {
  container.requestPointerLock();
  if (audioCtx.state === 'suspended') audioCtx.resume();
});

const getInput = (): FlightInput => {
  const st = inputMgr.state;
  const sens = 0.0022;
  flight.aimYaw -= st.mouseX * sens;
  flight.aimPitch = THREE.MathUtils.clamp(flight.aimPitch - st.mouseY * sens, -85 * Math.PI/180, 85 * Math.PI/180);
  st.mouseX = 0; st.mouseY = 0;
  
  return {
    thrust: st.keys.KeyW ? 1 : 0,
    boost: !!st.keys.ShiftLeft,
    brake: !!st.keys.KeyS,
    up: (st.keys.Space ? 1 : 0) - (st.keys.ControlLeft ? 1 : 0),
    strafe: (st.keys.KeyD ? 1 : 0) - (st.keys.KeyA ? 1 : 0),
    aimYaw: flight.aimYaw,
    aimPitch: flight.aimPitch,
  };
};

const loop = createLoop((fixedDt, frameDt, timeScale) => {
  if (fixedDt > 0) {
    flight.update(fixedDt, getInput(), { bounds: new THREE.Box3() });
    heroEntity.group.position.copy(flight.pos);
    heroEntity.group.quaternion.copy(flight.quat);
  }
  if (frameDt > 0) {
    const aimDir = new THREE.Vector3(-Math.sin(flight.aimYaw), Math.sin(flight.aimPitch), -Math.cos(flight.aimYaw));
    safeNormalize(aimDir);
    camCtrl.update(frameDt, flight, aimDir, flight.vel.length());
    hud.update(flight.vel.length() * 3.6, flight.hp, "FLY");
    render(frameDt);
  }
});

loop.start();
```

### Tuning Tips
1. **Flight Momentum:** Adjust `CONFIG.Hero.TurnRateMin` and `TurnRateMax` to change how tightly the hero can corner at speed. Lower values mean heavier, wider turns.
2. **Boost Surge:** Modify `CONFIG.Hero.BoostMult` and `DragBoost` to alter the acceleration and top speed of the boost ability.
3. **Holding Difficulty:** Change `CONFIG.Holding.BrakeMax` and `StressRiseRate` to tune how quickly the airliner stops versus how easily its nose buckles under pressure.

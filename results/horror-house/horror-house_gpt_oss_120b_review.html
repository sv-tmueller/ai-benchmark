```js
import * as THREE from 'three';
import { EffectComposer } from 'three/examples/jsm/postprocessing/EffectComposer.js';
import { RenderPass } from 'three/examples/jsm/postprocessing/RenderPass.js';
import { UnrealBloomPass } from 'three/examples/jsm/postprocessing/UnrealBloomPass.js';
import Game from './Game.js';
import HUD from './ui/HUD.js';

let renderer, scene, camera, composer, game, hud;

init();

function init() {
  const canvas = document.getElementById('glCanvas');
  renderer = new THREE.WebGLRenderer({ canvas, antialias: true });
  renderer.setSize(window.innerWidth, window.innerHeight);
  renderer.setPixelRatio(window.devicePixelRatio);
  renderer.shadowMap.enabled = true;
  renderer.shadowMap.type = THREE.PCFSoftShadowMap;

  scene = new THREE.Scene();
  camera = new THREE.PerspectiveCamera(75, innerWidth / innerHeight, 0.1, 100);
  camera.position.set(0, 1.6, 5);

  // post‑processing
  composer = new EffectComposer(renderer);
  composer.addPass(new RenderPass(scene, camera));
  const bloom = new UnrealBloomPass(new THREE.Vector2(innerWidth, innerHeight), 0.3, 0.4, 0.85);
  composer.addPass(bloom);

  // UI / HUD
  hud = new HUD();

  // Game instance
  game = new Game({ scene, camera, renderer, composer, hud });

  // title screen start
  document.getElementById('titleScreen').addEventListener('click', () => {
    document.getElementById('titleScreen').classList.add('hidden');
    game.start();
  });

  // resize handling
  window.addEventListener('resize', onWindowResize);
}

function onWindowResize() {
  const w = innerWidth, h = innerHeight;
  camera.aspect = w / h;
  camera.updateProjectionMatrix();
  renderer.setSize(w, h);
  composer.setSize(w, h);
}

// animation loop
function animate() {
  requestAnimationFrame(animate);
  const dt = Math.min(0.033, renderer.info.render.frame / 60);
  game.update(dt);
  composer.render();
}
animate();
```

## `src/Game.js`
```js
import Player from './player/Player.js';
import House from './world/House.js';
import AudioEngine from './systems/AudioEngine.js';
import Ghost from './systems/Ghost.js';
import ScareDirector from './systems/ScareDirector.js';

export default class Game {
  constructor({ scene, camera, renderer, composer, hud }) {
    this.scene = scene;
    this.camera = camera;
    this.renderer = renderer;
    this.composer = composer;
    this.hud = hud;

    this.state = 'TITLE'; // TITLE, PLAYING, CAUGHT, ESCAPED
    this.startTime = 0;
    this.keysCollected = 0;
    this.burnCount = 0;

    // core systems
    this.player = new Player(camera, scene, hud);
    this.house = new House(scene);
    this.audio = new AudioEngine(this.scene);
    this.ghost = new Ghost(scene, this.player);
    this.scareDirector = new ScareDirector(scene, this.player, this.ghost, this.hud);
  }

  start() {
    this.state = 'PLAYING';
    this.startTime = performance.now();
    this.player.enableControls();
    this.audio.startAmbient();
    this.hud.showHUD(true);
  }

  restart() {
    // simple full reload – satisfies “fast respawn”
    location.reload();
  }

  win() {
    this.state = 'ESCAPED';
    const timeSec = ((performance.now() - this.startTime) / 1000).toFixed(1);
    document.getElementById('stats').innerText = `Time: ${timeSec}s – Banishes: ${this.burnCount}`;
    document.getElementById('winScreen').classList.remove('hidden');
    this.player.disableControls();
    this.audio.stopAll();
  }

  die() {
    this.state = 'CAUGHT';
    document.getElementById('deathScreen').classList.remove('hidden');
    this.player.disableControls();
    this.audio.playDeathScream();
  }

  update(dt) {
    if (this.state !== 'PLAYING') return;

    this.player.update(dt);
    this.ghost.update(dt);
    this.scareDirector.update(dt);

    // key collection
    const key = this.player.checkPickup('key');
    if (key) {
      this.keysCollected++;
      this.hud.setKeyCount(this.keysCollected);
      if (this.keysCollected === 3) this.house.unlockFrontDoor();
    }

    // battery UI
    this.hud.setBattery(this.player.battery);

    // burn meter UI
    if (this.ghost.isBurning) {
      this.hud.showBurnMeter(true);
      this.hud.setBurn(this.ghost.burnProgress);
    } else {
      this.hud.showBurnMeter(false);
    }

    // win condition
    if (this.house.isFrontDoorOpen && this.keysCollected === 3) {
      this.win();
    }

    // death condition (ghost contact)
    if (this.ghost.isTouchingPlayer()) {
      this.die();
    }
  }
}
```

```js
// Procedural canvas textures – no external files
export const Textures = {
  wood() {
    const size = 256;
    const canvas = document.createElement('canvas');
    canvas.width = canvas.height = size;
    const ctx = canvas.getContext('2d');

    // base wood colour
    ctx.fillStyle = '#4a2e19';
    ctx.fillRect(0, 0, size, size);

    // noise
    const img = ctx.getImageData(0, 0, size, size);
    for (let i = 0; i < img.data.length; i += 4) {
      const v = (Math.random() - 0.5) * 30;
      img.data[i] = img.data[i] + v;
      img.data[i + 1] = img.data[i + 1] + v * 0.8;
      img.data[i + 2] = img.data[i + 2] + v * 0.6;
    }
    ctx.putImageData(img, 0, 0);
    return new THREE.CanvasTexture(canvas);
  },

  wallpaper() {
    const size = 256;
    const canvas = document.createElement('canvas');
    canvas.width = canvas.height = size;
    const ctx = canvas.getContext('2d');

    ctx.fillStyle = '#2b1c1c';
    ctx.fillRect(0, 0, size, size);
    ctx.strokeStyle = '#3a2a2a';
    ctx.lineWidth = 2;
    for (let i = 0; i < 20; i++) {
      ctx.beginPath();
      ctx.moveTo(Math.random() * size, Math.random() * size);
      ctx.lineTo(Math.random() * size, Math.random() * size);
      ctx.stroke();
    }
    return new THREE.CanvasTexture(canvas);
  },

  face() {
    const size = 128;
    const canvas = document.createElement('canvas');
    canvas.width = canvas.height = size;
    const ctx = canvas.getContext('2d');

    // background
    ctx.fillStyle = '#111';
    ctx.fillRect(0, 0, size, size);

    // eyes
    ctx.fillStyle = '#222';
    ctx.fillRect(30, 40, 20, 20);
    ctx.fillRect(78, 40, 20, 20);

    // mouth (randomly choose one of three)
    const r = Math.random();
    ctx.fillStyle = '#333';
    if (r < 0.33) {
      ctx.fillRect(45, 80, 40, 10);
    } else if (r < 0.66) {
      ctx.beginPath();
      ctx.moveTo(45, 85);
      ctx.quadraticCurveTo(65, 70, 85, 85);
      ctx.lineTo(85, 90);
      ctx.quadraticCurveTo(65, 75, 45, 90);
      ctx.closePath();
      ctx.fill();
    } else {
      ctx.arc(65, 85, 20, Math.PI, 0, false);
      ctx.fill();
    }
    return new THREE.CanvasTexture(canvas);
  },

  // simple dust particle texture
  dust() {
    const size = 64;
    const canvas = document.createElement('canvas');
    canvas.width = canvas.height = size;
    const ctx = canvas.getContext('2d');
    const grad = ctx.createRadialGradient(size/2, size/2, 0, size/2, size/2, size/2);
    grad.addColorStop(0, 'rgba(255,255,255,0.8)');
    grad.addColorStop(1, 'rgba(255,255,255,0)');
    ctx.fillStyle = grad;
    ctx.fillRect(0,0,size,size);
    return new THREE.CanvasTexture(canvas);
  }
};
```

```js
import * as THREE from 'three';
import { Textures } from './Textures.js';

export default class House {
  constructor(scene) {
    this.scene = scene;
    this.isFrontDoorOpen = false;
    this.build();
  }

  build() {
    const floorMat = new THREE.MeshStandardMaterial({ map: Textures.wood() });
    const wallMat = new THREE.MeshStandardMaterial({ map: Textures.wallpaper() });

    // floor
    const floor = new THREE.Mesh(
      new THREE.PlaneGeometry(20, 16),
      floorMat
    );
    floor.rotation.x = -Math.PI / 2;
    floor.receiveShadow = true;
    this.scene.add(floor);

    // walls (simple box)
    const wallHeight = 3;
    const wallThickness = 0.2;
    const wallGeo = new THREE.BoxGeometry(20, wallHeight, wallThickness);
    const backWall = new THREE.Mesh(wallGeo, wallMat);
    backWall.position.set(0, wallHeight/2, -8);
    backWall.receiveShadow = true;
    backWall.castShadow = true;
    this.scene.add(backWall);

    const frontWall = new THREE.Mesh(wallGeo, wallMat);
    frontWall.position.set(0, wallHeight/2, 8);
    frontWall.receiveShadow = true;
    frontWall.castShadow = true;
    this.scene.add(frontWall);

    const leftWall = new THREE.Mesh(
      new THREE.BoxGeometry(wallThickness, wallHeight, 16),
      wallMat
    );
    leftWall.position.set(-10, wallHeight/2, 0);
    leftWall.receiveShadow = true;
    leftWall.castShadow = true;
    this.scene.add(leftWall);

    const rightWall = leftWall.clone();
    rightWall.position.set(10, wallHeight/2, 0);
    this.scene.add(rightWall);

    // doors (simple thin boxes with hinge animation later)
    this.frontDoor = this.createDoor(new THREE.Vector3(0, 1.5, 8.1));
    this.studyDoor = this.createDoor(new THREE.Vector3(-6, 1.5, -4));
    this.bedroomDoor = this.createDoor(new THREE.Vector3(6, 1.5, -4));

    // items (keys, batteries, notes)
    this.items = [];

    // keys – random positions within three rooms
    const keyRooms = [
      { name: 'kitchen', pos: new THREE.Vector3(4, 0.5, -2) },
      { name: 'study',   pos: new THREE.Vector3(-6, 0.5, -2) },
      { name: 'bedroom', pos: new THREE.Vector3(6, 0.5, -2) }
    ];
    this.shuffleArray(keyRooms);
    keyRooms.forEach((room, i) => {
      const key = this.createItem('key', room.pos);
      key.userData.id = i + 1;
      this.items.push(key);
    });

    // batteries – fixed
    const batPos = [
      new THREE.Vector3(-4,0.5,0),
      new THREE.Vector3(4,0.5,0),
      new THREE.Vector3(0,0.5,-6),
      new THREE.Vector3(0,0.5,6)
    ];
    batPos.forEach(p=> this.items.push(this.createItem('battery', p)));

    // notes – simple planes with text
    const note = this.createNote(
      new THREE.Vector3(0,1,7.5),
      "Find the three brass keys. The front door won't open without them."
    );
    this.scene.add(note);
  }

  createDoor(position) {
    const geo = new THREE.BoxGeometry(1, 2.5, 0.1);
    const mat = new THREE.MeshStandardMaterial({ color: 0x332211 });
    const door = new THREE.Mesh(geo, mat);
    door.position.copy(position);
    door.castShadow = true;
    door.receiveShadow = true;
    this.scene.add(door);
    // simple open flag
    door.userData.isOpen = false;
    return door;
  }

  createItem(type, position) {
    const geo = new THREE.SphereGeometry(0.15, 12, 12);
    const mat = new THREE.MeshStandardMaterial({
      color: type === 'key' ? 0xdaa520 : 0x555555,
      metalness: 0.8,
      roughness: 0.2
    });
    const mesh = new THREE.Mesh(geo, mat);
    mesh.position.copy(position);
    mesh.userData.type = type;
    mesh.castShadow = true;
    this.scene.add(mesh);
    return mesh;
  }

  createNote(position, text) {
    const canvas = document.createElement('canvas');
    canvas.width = 512;
    canvas.height = 256;
    const ctx = canvas.getContext('2d');
    ctx.fillStyle = '#111';
    ctx.fillRect(0,0,512,256);
    ctx.fillStyle = '#fff';
    ctx.font = '20px sans-serif';
    ctx.fillText(text, 20, 40);
    const tex = new THREE.CanvasTexture(canvas);
    const mat = new THREE.MeshBasicMaterial({ map: tex, side: THREE.DoubleSide });
    const plane = new THREE.Mesh(new THREE.PlaneGeometry(2,1), mat);
    plane.position.copy(position);
    plane.userData.type = 'note';
    plane.userData.text = text;
    return plane;
  }

  unlockFrontDoor() {
    this.frontDoor.userData.isOpen = true;
    // simple animation – swing outward
    const tween = new TWEEN.Tween(this.frontDoor.rotation)
      .to({ y: -Math.PI/2 }, 800)
      .easing(TWEEN.Easing.Quadratic.Out)
      .start();
    this.isFrontDoorOpen = true;
  }

  // utility
  shuffleArray(arr) {
    for (let i = arr.length-1; i>0; i--) {
      const j = Math.floor(Math.random()*(i+1));
      [arr[i],arr[j]] = [arr[j],arr[i]];
    }
  }
}
```

## `src/player/Player.js`
```js
import * as THREE from 'three';
import { Textures } from '../world/Textures.js';

export default class Player {
  constructor(camera, scene, hud) {
    this.camera = camera;
    this.scene = scene;
    this.hud = hud;

    // movement
    this.velocity = new THREE.Vector3();
    this.direction = new THREE.Vector3();
    this.speed = 3;
    this.sprintMultiplier = 1.5;
    this.canMove = false;

    // battery
    this.maxBattery = 100;
    this.battery = this.maxBattery;
    this.batteryDrainRate = 4; // % per minute
    this.burnDrainMultiplier = 3;

    // flashlight
    this.flashlight = new THREE.SpotLight(0xffffff, 2, 15, Math.PI/6, 0.5, 0.5);
    this.flashlight.position.set(0,0,0);
    this.flashlight.target.position.set(0,0,-1);
    this.flashlight.castShadow = true;
    this.flashlight.shadow.mapSize.width = 1024;
    this.flashlight.shadow.mapSize.height = 1024;
    this.flashlight.shadow.bias = -0.0005;
    this.flashlight.shadow.radius = 2;
    this.flashlight.shadow.camera.near = 0.1;
    this.flashlight.shadow.camera.far = 20;
    this.scene.add(this.flashlight);
    this.scene.add(this.flashlight.target);

    // lens cookie (simple radial gradient)
    const cookie = new THREE.CanvasTexture(this.makeCookieCanvas());
    this.flashlight.map = cookie;

    // pointer lock
    this.initPointerLock();

    // raycaster for interaction
    this.raycaster = new THREE.Raycaster();

    // bind
    this.onKeyDown = this.onKeyDown.bind(this);
    this.onKeyUp = this.onKeyUp.bind(this);
    this.update = this.update.bind(this);
  }

  makeCookieCanvas() {
    const size = 256;
    const canvas = document.createElement('canvas');
    canvas.width = canvas.height = size;
    const ctx = canvas.getContext('2d');
    const grad = ctx.createRadialGradient(size/2,size/2,0,size/2,size/2,size/2);
    grad.addColorStop(0,'rgba(255,255,255,1)');
    grad.addColorStop(1,'rgba(255,255,255,0)');
    ctx.fillStyle = grad;
    ctx.fillRect(0,0,size,size);
    return canvas;
  }

  initPointerLock() {
    const canvas = document.getElementById('glCanvas');
    canvas.requestPointerLock = canvas.requestPointerLock ||
                                canvas.mozRequestPointerLock;
    document.exitPointerLock = document.exitPointerLock ||
                               document.mozExitPointerLock;

    canvas.addEventListener('click', () => {
      canvas.requestPointerLock();
    });

    const onPointerLockChange = () => {
      if (document.pointerLockElement === canvas) {
        document.addEventListener('mousemove', this.onMouseMove);
        this.canMove = true;
      } else {
        document.removeEventListener('mousemove', this.onMouseMove);
        this.canMove = false;
      }
    };
    document.addEventListener('pointerlockchange', onPointerLockChange);
    document.addEventListener('keydown', this.onKeyDown);
    document.addEventListener('keyup', this.onKeyUp);
  }

  onMouseMove = (event) => {
    const movementX = event.movementX || 0;
    const movementY = event.movementY || 0;
    const euler = new THREE.Euler(0,0,0,'YXZ');
    euler.setFromQuaternion(this.camera.quaternion);
    const PI_2 = Math.PI / 2;
    euler.y -= movementX * 0.002;
    euler.x -= movementY * 0.002;
    euler.x = Math.max(-PI_2, Math.min(PI_2, euler.x));
    this.camera.quaternion.setFromEuler(euler);
  };

  onKeyDown(event) {
    switch(event.code) {
      case 'KeyW': this.moveForward = true; break;
      case 'KeyS': this.moveBackward = true; break;
      case 'KeyA': this.moveLeft = true; break;
      case 'KeyD': this.moveRight = true; break;
      case 'ShiftLeft':
      case 'ShiftRight': this.isSprinting = true; break;
      case 'KeyE': this.interact(); break;
    }
  }

  onKeyUp(event) {
    switch(event.code) {
      case 'KeyW': this.moveForward = false; break;
      case 'KeyS': this.moveBackward = false; break;
      case 'KeyA': this.moveLeft = false; break;
      case 'KeyD': this.moveRight = false; break;
      case 'ShiftLeft':
      case 'ShiftRight': this.isSprinting = false; break;
    }
  }

  enableControls() {
    this.canMove = true;
  }

  disableControls() {
    this.canMove = false;
  }

  update(dt) {
    if (!this.canMove) return;

    // battery drain
    const drain = this.batteryDrainRate * dt * (this.isBurning ? this.burnDrainMultiplier : 1);
    this.battery = Math.max(0, this.battery - drain);
    if (this.battery === 0) this.flashlight.intensity = 0;
    else this.flashlight.intensity = 2 * (this.battery / this.maxBattery);

    // movement
    this.direction.set(0,0,0);
    if (this.moveForward) this.direction.z -= 1;
    if (this.moveBackward) this.direction.z += 1;
    if (this.moveLeft) this.direction.x -= 1;
    if (this.moveRight) this.direction.x += 1;
    this.direction.normalize();

    const actualSpeed = this.speed * (this.isSprinting ? this.sprintMultiplier : 1);
    const move = this.direction.clone().applyQuaternion(this.camera.quaternion).multiplyScalar(actualSpeed * dt);
    this.camera.position.add(move);

    // flashlight follows camera
    this.flashlight.position.copy(this.camera.position);
    const target = new THREE.Vector3(0,0,-1).applyQuaternion(this.camera.quaternion).add(this.camera.position);
    this.flashlight.target.position.copy(target);
    this.flashlight.target.updateMatrixWorld();

    // head bob (simple)
    const bobSpeed = 8;
    const bobAmount = 0.02;
    const time = performance.now() * 0.001;
    this.camera.position.y = 1.6 + Math.sin(time * bobSpeed) * bobAmount * (this.moveForward||this.moveBackward||this.moveLeft||this.moveRight ? 1 : 0);
  }

  interact() {
    // raycast forward 2 meters
    this.raycaster.set(this.camera.position, new THREE.Vector3(0,0,-1).applyQuaternion(this.camera.quaternion));
    const intersects = this.raycaster.intersectObjects(this.scene.children, true);
    if (intersects.length === 0) return;
    const obj = intersects[0].object;
    if (obj.userData.type === 'key' && intersects[0].distance < 2) {
      this.scene.remove(obj);
      this.hud.showMessage('Picked up a key');
      this.lastPickup = {type:'key', object:obj};
    } else if (obj.userData.type === 'battery' && intersects[0].distance < 2) {
      this.scene.remove(obj);
      this.battery = Math.min(this.maxBattery, this.battery + 30);
      this.hud.showMessage('Battery collected');
    } else if (obj.userData.type === 'note' && intersects[0].distance < 2) {
      this.hud.showNote(obj.userData.text);
    } else if (obj.userData.isOpen === false && intersects[0].distance < 2) {
      // door interaction
      obj.userData.isOpen = true;
      // simple swing animation
      const targetRot = obj.rotation.y - Math.PI/2;
      new TWEEN.Tween(obj.rotation)
        .to({ y: targetRot }, 600)
        .easing(TWEEN.Easing.Quadratic.Out)
        .start();
      this.hud.showMessage('Door opened');
    }
  }

  // called by Game to see if a key was just taken
  checkPickup(type) {
    if (this.lastPickup && this.lastPickup.type === type) {
      const p = this.lastPickup;
      this.lastPickup = null;
      return p.object;
    }
    return null;
  }
}
```

```js
import * as THREE from 'three';

export default class AudioEngine {
  constructor(scene) {
    this.scene = scene;
    this.context = new (window.AudioContext || window.webkitAudioContext)();
    this.masterGain = this.context.createGain();
    this.masterGain.connect(this.context.destination);
    this.masterGain.gain.value = 0.8;

    this.nodes = {};

    // ambient drone (low rumble)
    this.createDrone('ambient', 30, 0.2);
    // rain loop (white noise filtered)
    this.createRain();
    // thunder scheduler
    this.thunderTimer = 0;
  }

  createDrone(name, freq, gain) {
    const osc = this.context.createOscillator();
    osc.type = 'sine';
    osc.frequency.value = freq;
    const g = this.context.createGain();
    g.gain.value = gain;
    osc.connect(g).connect(this.masterGain);
    osc.start();
    this.nodes[name] = {osc,g};
  }

  createRain() {
    const bufferSize = 2 * this.context.sampleRate;
    const noise = this.context.createBufferSource();
    const data = new Float32Array(bufferSize);
    for (let i=0;i<bufferSize;i++) data[i] = Math.random()*2-1;
    const buf = this.context.createBuffer(1, bufferSize, this.context.sampleRate);
    buf.getChannelData(0).set(data);
    noise.buffer = buf;
    noise.loop = true;

    const filter = this.context.createBiquadFilter();
    filter.type = 'highpass';
    filter.frequency.value = 2000;

    const gain = this.context.createGain();
    gain.gain.value = 0.1;

    noise.connect(filter).connect(gain).connect(this.masterGain);
    noise.start();
    this.nodes.rain = {noise,filter,gain};
  }

  startAmbient() {
    // already started in constructor
  }

  stopAll() {
    Object.values(this.nodes).forEach(n => {
      if (n.osc) n.osc.stop();
      if (n.noise) n.noise.stop();
    });
  }

  // thunder + lightning flash
  triggerThunder() {
    const osc = this.context.createOscillator();
    osc.type = 'sawtooth';
    osc.frequency.value = 80;
    const gain = this.context.createGain();
    gain.gain.setValueAtTime(0, this.context.currentTime);
    gain.gain.linearRampToValueAtTime(1, this.context.currentTime + 0.01);
    gain.gain.exponentialRampToValueAtTime(0.001, this.context.currentTime + 1.2);
    osc.connect(gain).connect(this.masterGain);
    osc.start(this.context.currentTime);
    osc.stop(this.context.currentTime + 1.3);
  }

  // death scream (single burst)
  playDeathScream() {
    const osc = this.context.createOscillator();
    osc.type = 'square';
    osc.frequency.setValueAtTime(200, this.context.currentTime);
    osc.frequency.exponentialRampToValueAtTime(50, this.context.currentTime + 0.5);
    const gain = this.context.createGain();
    gain.gain.setValueAtTime(0, this.context.currentTime);
    gain.gain.linearRampToValueAtTime(1, this.context.currentTime + 0.05);
    gain.gain.exponentialRampToValueAtTime(0.001, this.context.currentTime + 0.8);
    osc.connect(gain).connect(this.masterGain);
    osc.start(this.context.currentTime);
    osc.stop(this.context.currentTime + 1);
  }

  // call each frame to possibly fire lightning
  update(dt) {
    this.thunderTimer -= dt;
    if (this.thunderTimer <= 0) {
      if (Math.random() < 0.03) { // occasional lightning
        this.triggerThunder();
        // flash effect via CSS
        const grain = document.getElementById('grain');
        grain.style.transition = 'background 0.1s';
        grain.style.background = 'rgba(255,255,255,0.6)';
        setTimeout(()=> grain.style.background='rgba(255,255,255,0.15)', 100);
      }
      this.thunderTimer = 5 + Math.random()*10;
    }
  }
}
```

```js
import * as THREE from 'three';
import { Textures } from '../world/Textures.js';

export default class Ghost {
  constructor(scene, player) {
    this.scene = scene;
    this.player = player;
    this.isBurning = false;
    this.burnProgress = 0;
    this.burnTime = 2; // seconds to fill meter
    this.respawnDelay = 8; // seconds after banish

    // visual
    const geo = new THREE.CylinderGeometry(0.5,0.5,2,8);
    const mat = new THREE.MeshStandardMaterial({
      color: 0x111111,
      emissive: 0x550000,
      transparent: true,
      opacity: 0.7,
      side: THREE.DoubleSide
    });
    this.mesh = new THREE.Mesh(geo, mat);
    this.mesh.position.set(0,1, -5);
    this.mesh.visible = false;
    this.scene.add(this.mesh);

    // state machine
    this.state = 'HIDDEN'; // HIDDEN, APPEAR, STALK, CHASE, BANISHED, RESPAWN
    this.timer = 0;
  }

  appearAt(pos) {
    this.mesh.position.copy(pos);
    this.mesh.visible = true;
    this.state = 'APPEAR';
    this.timer = 0;
  }

  update(dt) {
    // burn handling
    if (this.isBurning) {
      this.burnProgress += dt / this.burnTime;
      if (this.burnProgress >= 1) {
        this.banish();
      }
    }

    // simple state machine
    this.timer += dt;
    switch(this.state) {
      case 'HIDDEN':
        // nothing
        break;
      case 'APPEAR':
        if (this.timer > 1) {
          this.state = 'HIDDEN';
          this.mesh.visible = false;
        }
        break;
      case 'STALK':
        // move slowly towards player, but stay partially hidden
        this.moveTowardsPlayer(0.5);
        if (Math.random() < 0.01) this.appearAt(this.randomSpot());
        break;
      case 'CHASE':
        this.moveTowardsPlayer(3);
        break;
      case 'BANISHED':
        if (this.timer > this.respawnDelay) {
          this.state = 'STALK';
          this.mesh.visible = true;
        }
        break;
    }
  }

  moveTowardsPlayer(speed) {
    const dir = new THREE.Vector3().subVectors(this.player.camera.position, this.mesh.position);
    dir.y = 0;
    dir.normalize();
    this.mesh.position.addScaledVector(dir, speed * 0.016);
  }

  randomSpot() {
    const x = (Math.random() - 0.5) * 18;
    const z = (Math.random() - 0.5) * 14;
    return new THREE.Vector3(x,1,z);
  }

  // called by ScareDirector when script wants a brief apparition
  triggerApparition() {
    this.appearAt(this.randomSpot());
  }

  // called when player holds flashlight on ghost
  startBurn() {
    this.isBurning = true;
    this.burnProgress = 0;
  }

  stopBurn() {
    this.isBurning = false;
    this.burnProgress = 0;
  }

  banish() {
    this.isBurning = false;
    this.burnProgress = 0;
    this.state = 'BANISHED';
    this.timer = 0;
    this.mesh.visible = false;
    // brief shriek sound (synth)
    const ctx = this.player.audio.context;
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();
    osc.type = 'sawtooth';
    osc.frequency.setValueAtTime(200, ctx.currentTime);
    osc.frequency.exponentialRampToValueAtTime(50, ctx.currentTime + 0.4);
    gain.gain.setValueAtTime(0, ctx.currentTime);
    gain.gain.linearRampToValueAtTime(0.7, ctx.currentTime + 0.05);
    gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.6);
    osc.connect(gain).connect(this.player.audio.masterGain);
    osc.start(ctx.currentTime);
    osc.stop(ctx.currentTime + 0.7);
  }

  isTouchingPlayer() {
    const dist = this.mesh.position.distanceTo(this.player.camera.position);
    return dist < 0.8 && this.state === 'CHASE';
  }

  // external trigger to start chase (final key)
  startChase() {
    this.state = 'CHASE';
    this.mesh.visible = true;
    this.mesh.position.copy(this.randomSpot());
  }
}
```

```js
import * as THREE from 'three';
import { Textures } from '../world/Textures.js';

export default class ScareDirector {
  constructor(scene, player, ghost, hud) {
    this.scene = scene;
    this.player = player;
    this.ghost = ghost;
    this.hud = hud;

    this.triggers = [];
    this.initTriggers();

    this.ambientTimer = 0;
    this.ambientInterval = 5 + Math.random()*10;
  }

  initTriggers() {
    // each trigger is a Box3 + callback + fired flag
    const add = (min, max, cb) => {
      const box = new THREE.Box3(new THREE.Vector3(...min), new THREE.Vector3(...max));
      this.triggers.push({box, cb, fired:false});
    };
    // 1. living room entry
    add([-5,0,-2], [5,3,2], () => {
      this.playAmbient('thump');
    });
    // 2. hallway first time
    add([-2,0,-4], [2,3,-2], () => {
      this.flashLights();
    });
    // 3. kitchen cabinet
    add([3,0,-6], [5,3,-4], () => {
      this.playAmbient('cabinetBang');
    });
    // 4. study key apparition
    add([-7,0,-3], [-5,3,-1], () => {
      this.ghost.triggerApparition();
    });
    // 5. bedroom door slam
    add([5,0,-3], [7,3,-1], () => {
      this.playAmbient('doorSlam');
    });
    // 6. after 2nd key – start stalking
    // handled in Game when key count reaches 2
    // 7. final key – blackout & chase
    // handled in Game when key count reaches 3
  }

  update(dt) {
    // ambient random events
    this.ambientTimer += dt;
    if (this.ambientTimer > this.ambientInterval) {
      this.randomAmbient();
      this.ambientTimer = 0;
      this.ambientInterval = 5 + Math.random()*10;
    }

    // check triggers
    const pos = this.player.camera.position;
    this.triggers.forEach(t => {
      if (!t.fired && t.box.containsPoint(pos)) {
        t.fired = true;
        t.cb();
      }
    });
  }

  flashLights() {
    // quick flicker via CSS overlay
    const grain = document.getElementById('grain');
    grain.style.transition = 'background 0.05s';
    grain.style.background = 'rgba(255,255,255,0.4)';
    setTimeout(()=> grain.style.background='rgba(255,255,255,0.15)', 100);
    // play slam sound (synth)
    const ctx = this.player.audio.context;
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();
    osc.type = 'square';
    osc.frequency.value = 120;
    gain.gain.setValueAtTime(0, ctx.currentTime);
    gain.gain.linearRampToValueAtTime(0.8, ctx.currentTime + 0.02);
    gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.4);
    osc.connect(gain).connect(this.player.audio.masterGain);
    osc.start(ctx.currentTime);
    osc.stop(ctx.currentTime + 0.5);
  }

  playAmbient(name) {
    // placeholder: just a short synth burst per name
    const ctx = this.player.audio.context;
    const osc = ctx.createOscillator();
    const gain = ctx.createGain();
    osc.type = 'sine';
    osc.frequency.value = name === 'thump' ? 60 : 80;
    gain.gain.setValueAtTime(0, ctx.currentTime);
    gain.gain.linearRampToValueAtTime(0.5, ctx.currentTime + 0.01);
    gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 0.6);
    osc.connect(gain).connect(this.player.audio.masterGain);
    osc.start(ctx.currentTime);
    osc.stop(ctx.currentTime + 0.7);
  }

  randomAmbient() {
    const events = ['creak','whisper','knock'];
    const ev = events[Math.floor(Math.random()*events.length)];
    this.playAmbient(ev);
  }
}
```
